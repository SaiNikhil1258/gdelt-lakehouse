import argparse
import io
import time
import zipfile
import boto3
import requests

MANIFEST_URL = "http://data.gdeltproject.org/gdeltv2/lastupdate.txt"
BASE_URL = "http://data.gdeltproject.org/gdeltv2"

STREAM_SUFFIX = {
    "events": "export.CSV.zip",
    "mentions": "mentions.CSV.zip",
    "gkg": "gkg.csv.zip",
}

def get_latest_timestamp() -> str:
    resp = requests.get(MANIFEST_URL, timeout=30)
    resp.raise_for_status()
    first_line = resp.text.strip().split("\n")[0]
    _, _, url = first_line.split()
    return url.rsplit("/", 1)[-1].split(".")[0]

def s3_client():
    return boto3.client(
        "s3",
        endpoint_url="http://localhost:9000",
        aws_access_key_id="admin",
        aws_secret_access_key="changeme123",
    )

def fetch_and_upload(timestamp: str, stream: str, client, max_retries=3, backoff_base=30):
    url = f"{BASE_URL}/{timestamp}.{STREAM_SUFFIX[stream]}"

    for attempt in range(1, max_retries + 1):
        resp = requests.get(url, timeout=60)

        if resp.status_code == 404:
            if attempt < max_retries:
                wait = backoff_base * attempt   # 30s, 60s, 90s
                print(f"  [{stream}] 404 on attempt {attempt}/{max_retries} — "
                      f"GDELT may still be processing this slot, retrying in {wait}s")
                time.sleep(wait)
                continue
            print(f"  [{stream}] SKIPPED after {max_retries} attempts — "
                  f"this slot genuinely has no {stream} file")
            return False

        resp.raise_for_status()
        with zipfile.ZipFile(io.BytesIO(resp.content)) as z:
            data = z.read(z.namelist()[0])

        key = f"{stream}/{timestamp}.csv"
        client.put_object(Bucket="gdelt-landing", Key=key, Body=data)
        print(f"  [{stream}] uploaded s3://gdelt-landing/{key} ({len(data):,} bytes)")
        return True

    return False

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--timestamp", help="YYYYMMDDHHMMSS, defaults to latest available")
    parser.add_argument("--streams", nargs="+", default=["events", "mentions", "gkg"])
    args = parser.parse_args()

    ts = args.timestamp or get_latest_timestamp()
    print(f"Batch timestamp: {ts}")

    client = s3_client()
    results = {stream: fetch_and_upload(ts, stream, client) for stream in args.streams}

    failed = [s for s, ok in results.items() if not ok]
    if failed:
        print(f"\nIncomplete batch — missing: {failed}")
    else:
        print(f"\nComplete batch: {ts}")

# spark/jobs/bronze_ingest.py
import argparse
from pyspark.sql import SparkSession, functions as F
from pyspark.sql.types import StructType
from schemas import EVENTS_COLUMNS, MENTIONS_COLUMNS, GKG_COLUMNS
from utils import parse_gdelt_ts


def dedup_and_append(spark, batch_df, target_table: str, key_cols: list):
    """Insert-only idempotency via an explicit left-anti join, since Iceberg's
    MERGE INTO insert-only optimization was observed to append duplicates
    unconditionally rather than checking existing keys."""
    existing_keys = spark.table(target_table).select(*key_cols).distinct()
    new_rows = batch_df.join(existing_keys, on=key_cols, how="left_anti")
    new_count = new_rows.count()
    total_count = batch_df.count()
    if new_count > 0:
        new_rows.writeTo(target_table).append()
    return new_count, total_count


def ingest_events_batch(spark, path: str):
    print(f"Reading events from: {path}")
    target_table = "local.bronze.events"
    target_schema = spark.table(target_table).schema
    
    # Map exact CSV column order to Iceberg data types
    csv_schema = StructType([target_schema[col] for col in EVENTS_COLUMNS if col in target_schema.names])

    raw_df = spark.read \
        .option("sep", "\t") \
        .option("nullValue", "") \
        .schema(csv_schema) \
        .csv(path)

    batch_df = raw_df.withColumn("date_added_ts", parse_gdelt_ts("DATEADDED"))
    return dedup_and_append(spark, batch_df, target_table, ["GLOBALEVENTID"])


def ingest_mentions_batch(spark, path: str):
    print(f"Reading mentions from: {path}")
    target_table = "local.bronze.mentions"
    target_schema = spark.table(target_table).schema
    
    csv_schema = StructType([target_schema[col] for col in MENTIONS_COLUMNS if col in target_schema.names])

    raw_df = spark.read \
        .option("sep", "\t") \
        .option("nullValue", "") \
        .schema(csv_schema) \
        .csv(path)

    # NOTE: fmt below is a placeholder until diagnose_mentions.py output confirms the real format
    batch_df = raw_df.withColumn("mention_ts", parse_gdelt_ts("MentionTimeDate"))
    return dedup_and_append(
        spark, batch_df, target_table,
        ["GLOBALEVENTID", "MentionIdentifier", "mention_ts"]
    )


def ingest_gkg_batch(spark, path: str):
    print(f"Reading gkg from: {path}")
    target_table = "local.bronze.gkg"
    target_schema = spark.table(target_table).schema
    
    csv_schema = StructType([target_schema[col] for col in GKG_COLUMNS if col in target_schema.names])

    raw_df = spark.read \
        .option("sep", "\t") \
        .option("nullValue", "") \
        .schema(csv_schema) \
        .csv(path)

    batch_df = raw_df.withColumn("gkg_ts", parse_gdelt_ts("DATE"))
    return dedup_and_append(spark, batch_df, target_table, ["GKGRECORDID"])


STREAM_FUNCS = {
    "events": ingest_events_batch,
    "mentions": ingest_mentions_batch,
    "gkg": ingest_gkg_batch,
}

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--stream", required=True, choices=STREAM_FUNCS.keys())
    parser.add_argument("--path", required=True)
    args = parser.parse_args()

    spark = SparkSession.builder.appName(f"bronze-ingest-{args.stream}").getOrCreate()
    new_count, total_count = STREAM_FUNCS[args.stream](spark, args.path)
    print(f"{args.stream}: {new_count} new rows written, {total_count} rows in source batch "
          f"({total_count - new_count} already existed)")
    spark.stop()

if __name__ == "__main__":
    main()

from pyspark.sql import SparkSession, functions as F


def build_bilateral_sentiment(spark):
    events = spark.table("local.silver.events_india")

    gold = (
        events
        .withColumn(
            "partner_country",
            F.when(F.col("Actor1CountryCode") == "IND", F.col("Actor2CountryCode"))
             .otherwise(F.col("Actor1CountryCode"))
        )
        .filter(
            F.col("partner_country").isNotNull() &
            (F.col("partner_country") != "IND") &
            (F.col("partner_country") != "")
        )
        .groupBy("event_date", "partner_country")
        .agg(
            F.count("*").alias("event_count"),
            F.avg("GoldsteinScale").alias("avg_goldstein"),
            F.avg("AvgTone").alias("avg_tone"),
        )
    )

    row_count = gold.count()
    gold.writeTo("local.gold.india_bilateral_sentiment_daily").overwritePartitions()
    return row_count


def build_state_conflict(spark):
    events = spark.table("local.silver.events_india")

    gold = (
        events
        .filter(F.col("ActionGeo_ADM1Code").isNotNull())
        .groupBy("event_date", "ActionGeo_ADM1Code", "conflict_severity")
        .agg(
            F.count("*").alias("event_count"),
            F.avg("GoldsteinScale").alias("avg_goldstein"),
        )
        .withColumnRenamed("ActionGeo_ADM1Code", "adm1_code")
    )

    row_count = gold.count()
    gold.writeTo("local.gold.india_state_conflict_daily").overwritePartitions()
    return row_count


def merge_mention_counts(spark):
    """Upsert against the FULL current total for each event -- not an additive
    delta. See discussion above: Silver is a full recompute, not incremental,
    so this MERGE must replace each event's totals rather than add to them,
    or re-running the pipeline would double-count on every run."""
    mentions = spark.table("local.silver.mentions_enriched")

    agg = (
        mentions
        .groupBy("GLOBALEVENTID")
        .agg(
            F.count("*").alias("total_mentions"),
            F.countDistinct("MentionSourceName").alias("total_sources"),
            F.avg("MentionDocTone").alias("running_avg_tone"),
        )
    )
    agg.createOrReplaceTempView("mention_agg_source")

    spark.sql("""
        MERGE INTO local.gold.india_event_mention_counts t
        USING mention_agg_source s
        ON t.GLOBALEVENTID = s.GLOBALEVENTID
        WHEN MATCHED THEN UPDATE SET
            t.total_mentions    = s.total_mentions,
            t.total_sources     = s.total_sources,
            t.running_avg_tone  = s.running_avg_tone,
            t.last_updated_ts   = current_timestamp()
        WHEN NOT MATCHED THEN INSERT (
            GLOBALEVENTID, total_mentions, total_sources, running_avg_tone, last_updated_ts
        ) VALUES (
            s.GLOBALEVENTID, s.total_mentions, s.total_sources, s.running_avg_tone, current_timestamp()
        )
    """)

    return agg.count()


def main():
    spark = SparkSession.builder.appName("gold-aggregate").getOrCreate()

    b = build_bilateral_sentiment(spark)
    print(f"gold.india_bilateral_sentiment_daily: {b} rows written")

    s = build_state_conflict(spark)
    print(f"gold.india_state_conflict_daily: {s} rows written")

    m = merge_mention_counts(spark)
    print(f"gold.india_event_mention_counts: merged against {m} distinct events")

    spark.stop()

if __name__ == "__main__":
    main()

from pyspark.sql import SparkSession, functions as F
from utils import normalize_url


def transform_events_india(spark):
    bronze_events = spark.table("local.bronze.events")

    silver_events = (
        bronze_events
        .withColumn("event_date", F.to_date("SQLDATE", "yyyyMMdd"))
        .filter(
            (F.col("ActionGeo_CountryCode") == "IN") |
            (F.col("Actor1CountryCode") == "IND") |
            (F.col("Actor2CountryCode") == "IND")
        )
        .filter(F.col("SOURCEURL").isNotNull() & F.col("GLOBALEVENTID").isNotNull())
        .withColumn(
            "conflict_severity",
            F.when(F.col("QuadClass") == 4, "Material Conflict")
             .when(F.col("QuadClass") == 3, "Verbal Conflict")
             .when(F.col("QuadClass") == 2, "Material Cooperation")
             .otherwise("Verbal Cooperation")
        )
        .select(
            "GLOBALEVENTID", "event_date", "Actor1CountryCode", "Actor1Name",
            "Actor2CountryCode", "Actor2Name", "EventRootCode", "QuadClass",
            "conflict_severity", "GoldsteinScale", "AvgTone",
            "ActionGeo_CountryCode", "ActionGeo_ADM1Code", "ActionGeo_Lat", "ActionGeo_Long",
            "SOURCEURL"
        )
        .dropDuplicates(["GLOBALEVENTID"])  # defensive; Bronze should already guarantee this
    )

    row_count = silver_events.count()
    silver_events.writeTo("local.silver.events_india").overwritePartitions()
    return row_count


def transform_mentions_enriched(spark):
    bronze_mentions = spark.table("local.bronze.mentions")
    bronze_gkg = spark.table("local.bronze.gkg")
    silver_events = spark.table("local.silver.events_india").select("GLOBALEVENTID", "event_date")

    silver_mentions = (
        bronze_mentions.alias("m")
        .join(
            bronze_gkg.alias("g"),
            normalize_url(F.col("m.MentionIdentifier")) == normalize_url(F.col("g.DocumentIdentifier")),
            "left"
        )
        .join(
            silver_events.alias("e"),
            F.col("m.GLOBALEVENTID") == F.col("e.GLOBALEVENTID"),
            "inner"   # scope to India-relevant events only
        )
        .select(
            F.col("m.GLOBALEVENTID"), F.col("e.event_date"),
            F.col("m.MentionSourceName"), F.col("m.MentionIdentifier"),
            F.col("m.MentionDocTone"), F.col("m.Confidence"),
            F.col("g.V2Themes"), F.col("g.V2Tone")
        )
    )

    row_count = silver_mentions.count()
    silver_mentions.writeTo("local.silver.mentions_enriched").overwritePartitions()
    return row_count


def main():
    spark = SparkSession.builder.appName("silver-transform").getOrCreate()
    events_count = transform_events_india(spark)
    print(f"silver.events_india: {events_count} rows written")
    mentions_count = transform_mentions_enriched(spark)
    print(f"silver.mentions_enriched: {mentions_count} rows written")
    spark.stop()

if __name__ == "__main__":
    main()

"""Standalone check for MERGE INTO ... WHEN MATCHED THEN UPDATE, run once to
build confidence before Gold's incremental counters depend on it. Uses a
disposable scratch table -- doesn't touch bronze/silver/gold."""
from pyspark.sql import SparkSession

def main():
    spark = SparkSession.builder.appName("test-merge-update").getOrCreate()

    spark.sql("CREATE NAMESPACE IF NOT EXISTS local.scratch")
    spark.sql("DROP TABLE IF EXISTS local.scratch.merge_test")
    spark.sql("""
        CREATE TABLE local.scratch.merge_test (id BIGINT, val STRING)
        USING iceberg TBLPROPERTIES ('format-version'='2')
    """)
    spark.sql("INSERT INTO local.scratch.merge_test VALUES (1, 'original'), (2, 'original')")

    spark.createDataFrame([(1, "updated"), (3, "new")], ["id", "val"]) \
        .createOrReplaceTempView("merge_source")

    spark.sql("""
        MERGE INTO local.scratch.merge_test t
        USING merge_source s
        ON t.id = s.id
        WHEN MATCHED THEN UPDATE SET t.val = s.val
        WHEN NOT MATCHED THEN INSERT *
    """)

    rows = spark.sql("SELECT * FROM local.scratch.merge_test ORDER BY id").collect()
    actual = {row.id: row.val for row in rows}
    expected = {1: "updated", 2: "original", 3: "new"}

    print("Result after MERGE:", actual)
    print("PASS" if actual == expected else f"FAIL -- expected {expected}, got {actual}")

    spark.stop()

if __name__ == "__main__":
    main()

from pyspark.sql import functions as F

def normalize_url(col):
    """Lowercase, trim, strip query string — for the Mentions<->GKG soft join."""
    return F.lower(F.trim(F.regexp_replace(col, r"\?.*$", "")))

def parse_gdelt_ts(col, fmt="yyyyMMddHHmmss"):
    return F.to_timestamp(col, fmt)

def extract_batch_id(file_path: str) -> str:
    """20260910143000.export.CSV -> 20260910143000"""
    return file_path.split("/")[-1].split(".")[0]
