"""Download GDELT slot files into MinIO, then ingest them into the Bronze Iceberg tables.

Hardened version. Compared with the first one:
  * preflight: MinIO and the Spark master UI must be reachable
  * refuses to start if another backfill is already running (host lock file + the Spark
    master's list of running applications); override with --force
  * on SIGINT / SIGTERM it kills its own Spark application on the master, so an interrupted
    run no longer leaves an orphan job holding the cluster's cores
  * honest ending: BACKFILL COMPLETE only when nothing failed. Files GDELT answered 404 for
    are listed as warnings (exit 0, or exit 1 with --fail-on-missing)
  * exit code is non-zero whenever a download failed or a Spark chunk failed
"""
import argparse
import concurrent.futures
import fcntl
import io
import os
import signal
import subprocess
import sys
import time
import zipfile
from dataclasses import dataclass
from datetime import date, datetime, timedelta
from typing import Optional

import boto3
import requests
from botocore.config import Config
from botocore.exceptions import ClientError

# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------
BASE_URL = "http://data.gdeltproject.org/gdeltv2"
S3_ENDPOINT = os.getenv("GDELT_S3_ENDPOINT", "http://localhost:9000")
S3_BUCKET = os.getenv("GDELT_S3_BUCKET", "gdelt-landing")
AWS_ACCESS_KEY_ID = os.getenv("AWS_ACCESS_KEY_ID", "admin")
AWS_SECRET_ACCESS_KEY = os.getenv("AWS_SECRET_ACCESS_KEY", "changeme123")
AWS_REGION = os.getenv("AWS_REGION", "us-east-1")

SPARK_MASTER_UI = os.getenv("SPARK_MASTER_UI", "http://localhost:8080")
SPARK_APP_NAME = "gdelt-backfill-ingest"   # must match the appName in spark/jobs/backfill_ingest.py
LOCK_PATH = os.path.normpath(
    os.path.join(os.path.dirname(os.path.abspath(__file__)), os.pardir, ".backfill.lock")
)

STREAM_SUFFIX = {
    "events": "export.CSV.zip",
    "mentions": "mentions.CSV.zip",
    "gkg": "gkg.csv.zip",
}
DEFAULT_STREAMS = ["events", "mentions", "gkg"]
SLOT_MINUTES = 15
DEFAULT_WORKERS = 6
DEFAULT_RETRIES = 4
DEFAULT_CHUNK_DAYS = 7
REQUEST_TIMEOUT = 120
RETRY_BACKOFF = [15, 30, 60, 120]   # seconds between attempts for transient failures


# ---------------------------------------------------------------------------
# Data structures
# ---------------------------------------------------------------------------
@dataclass
class DownloadResult:
    timestamp: str
    stream: str
    status: str            # downloaded | already_exists | missing | failed
    bytes_written: int = 0
    error: Optional[str] = None


# ---------------------------------------------------------------------------
# Date / timestamp helpers
# ---------------------------------------------------------------------------
def parse_date(value: str) -> date:
    try:
        return datetime.strptime(value, "%Y%m%d").date()
    except ValueError:
        raise argparse.ArgumentTypeError(f"Invalid date '{value}'. Expected YYYYMMDD.")


def timestamp_range(start: date, end: date):
    """Every 15-minute GDELT timestamp in the inclusive date range (96 per day)."""
    current = datetime.combine(start, datetime.min.time())
    final = datetime.combine(end, datetime.max.time())
    while current <= final:
        yield current.strftime("%Y%m%d%H%M%S")
        current += timedelta(minutes=SLOT_MINUTES)


def split_into_chunks(start: date, end: date, chunk_days: int):
    """Split a long backfill into bounded date ranges, e.g. 09-01..09-30 with 7 -> 5 chunks."""
    current = start
    while current <= end:
        chunk_end = min(current + timedelta(days=chunk_days - 1), end)
        yield current, chunk_end
        current = chunk_end + timedelta(days=1)


# ---------------------------------------------------------------------------
# S3 / MinIO
# ---------------------------------------------------------------------------
def s3_client():
    return boto3.client(
        "s3",
        endpoint_url=S3_ENDPOINT,
        aws_access_key_id=AWS_ACCESS_KEY_ID,
        aws_secret_access_key=AWS_SECRET_ACCESS_KEY,
        region_name=AWS_REGION,
        config=Config(connect_timeout=10, read_timeout=60, retries={"max_attempts": 3}),
    )


def object_exists(client, key: str) -> bool:
    """Makes the downloader resumable: a file already in MinIO is never fetched again."""
    try:
        client.head_object(Bucket=S3_BUCKET, Key=key)
        return True
    except ClientError as exc:
        if exc.response.get("Error", {}).get("Code") in ("404", "NoSuchKey", "NotFound"):
            return False
        raise


# ---------------------------------------------------------------------------
# Spark master: running applications and cleanup
# ---------------------------------------------------------------------------
def running_backfill_apps():
    resp = requests.get(f"{SPARK_MASTER_UI}/json/", timeout=5)
    resp.raise_for_status()
    return [a for a in resp.json().get("activeapps", []) if a.get("name") == SPARK_APP_NAME]


def kill_backfill_apps() -> int:
    """Kill running backfill Spark applications via the master's kill endpoint
    (the same POST the master UI's '(kill)' link sends). Returns how many were targeted."""
    try:
        apps = running_backfill_apps()
    except requests.RequestException as exc:
        print(f"Could not reach the Spark master to stop the app: {exc}")
        return 0
    for app in apps:
        try:
            requests.post(
                f"{SPARK_MASTER_UI}/app/kill/",
                data={"id": app["id"], "terminate": "true"},
                timeout=10,
                allow_redirects=False,
            )
            print(f"Asked the master to kill {app['id']}")
        except requests.RequestException as exc:
            print(f"Could not kill {app['id']}: {exc}")
    return len(apps)


# ---------------------------------------------------------------------------
# Guards
# ---------------------------------------------------------------------------
def acquire_lock():
    """One backfill.py per project checkout. The OS drops the lock when the process dies."""
    handle = open(LOCK_PATH, "w")
    try:
        fcntl.flock(handle, fcntl.LOCK_EX | fcntl.LOCK_NB)
    except BlockingIOError:
        raise SystemExit(
            "Another backfill.py is already running here (lock held: "
            f"{LOCK_PATH}). Wait for it, or pass --force to override."
        )
    return handle


def preflight(args):
    try:
        s3_client().head_bucket(Bucket=S3_BUCKET)
    except Exception as exc:
        raise SystemExit(
            f"Preflight failed: cannot reach MinIO bucket '{S3_BUCKET}' at {S3_ENDPOINT} ({exc}).\n"
            "Is the stack up?  docker compose up -d spark-worker"
        )
    if args.download_only:
        return
    try:
        apps = running_backfill_apps()
    except requests.RequestException as exc:
        raise SystemExit(
            f"Preflight failed: cannot reach the Spark master UI at {SPARK_MASTER_UI} ({exc}).\n"
            "Is the stack up?  docker compose up -d spark-worker"
        )
    if apps and not args.force:
        ids = ", ".join(a["id"] for a in apps)
        raise SystemExit(
            f"A backfill Spark application is already running on the cluster ({ids}). "
            "Starting a second writer can duplicate rows. Wait for it, stop it from "
            f"{SPARK_MASTER_UI}, or pass --force."
        )


# ---------------------------------------------------------------------------
# GDELT download
# ---------------------------------------------------------------------------
def download_one(timestamp: str, stream: str, max_retries: int) -> DownloadResult:
    key = f"{stream}/{timestamp}.csv"
    url = f"{BASE_URL}/{timestamp}.{STREAM_SUFFIX[stream]}"
    client = s3_client()

    try:
        if object_exists(client, key):
            return DownloadResult(timestamp, stream, "already_exists")
    except Exception as exc:
        return DownloadResult(timestamp, stream, "failed", error=f"S3 HEAD failed: {exc}")

    for attempt in range(1, max_retries + 1):
        try:
            response = requests.get(url, timeout=REQUEST_TIMEOUT)

            # 404 is special for GDELT: the file may still be processing, or a slot may
            # genuinely never be published. Retry with backoff, then report 'missing'.
            if response.status_code == 404:
                if attempt < max_retries:
                    wait = RETRY_BACKOFF[min(attempt - 1, len(RETRY_BACKOFF) - 1)]
                    print(f"[{timestamp}] {stream}: 404 (attempt {attempt}/{max_retries}), retrying in {wait}s")
                    time.sleep(wait)
                    continue
                return DownloadResult(timestamp, stream, "missing", error="GDELT returned 404 after retries")

            response.raise_for_status()

            # Unzip one file in memory; workers are bounded, so memory is bounded.
            with zipfile.ZipFile(io.BytesIO(response.content)) as archive:
                members = archive.namelist()
                if not members:
                    raise RuntimeError("Downloaded ZIP contains no files")
                data = archive.read(members[0])

            client.put_object(Bucket=S3_BUCKET, Key=key, Body=data)
            return DownloadResult(timestamp, stream, "downloaded", bytes_written=len(data))

        except (requests.RequestException, zipfile.BadZipFile, ClientError) as exc:
            if attempt >= max_retries:
                return DownloadResult(timestamp, stream, "failed", error=str(exc))
            wait = RETRY_BACKOFF[min(attempt - 1, len(RETRY_BACKOFF) - 1)]
            print(f"[{timestamp}] {stream}: {type(exc).__name__} (attempt {attempt}/{max_retries}), retrying in {wait}s")
            time.sleep(wait)
        except Exception as exc:
            return DownloadResult(timestamp, stream, "failed", error=str(exc))

    return DownloadResult(timestamp, stream, "failed", error="Unknown download failure")


def download_chunk(start: date, end: date, streams: list, workers: int, retries: int):
    """Download one chunk with bounded parallelism.
    Returns (failed_results, missing_results, counts)."""
    tasks = [(ts, stream) for ts in timestamp_range(start, end) for stream in streams]
    total = len(tasks)

    print()
    print("=" * 72)
    print(f"Downloading {start} -> {end}")
    print(f"Slots: {total // len(streams)} | Streams: {', '.join(streams)} | Files: {total} | Workers: {workers}")
    print("=" * 72)

    counts = {"downloaded": 0, "already_exists": 0, "missing": 0, "failed": 0}
    failed, missing = [], []
    completed = 0

    executor = concurrent.futures.ThreadPoolExecutor(max_workers=workers)
    try:
        future_map = {
            executor.submit(download_one, ts, stream, retries): (ts, stream) for ts, stream in tasks
        }
        for future in concurrent.futures.as_completed(future_map):
            ts, stream = future_map[future]
            try:
                result = future.result()
            except Exception as exc:
                result = DownloadResult(ts, stream, "failed", error=str(exc))

            completed += 1
            counts[result.status] += 1
            if result.status == "failed":
                failed.append(result)
            elif result.status == "missing":
                missing.append(result)

            if result.status == "downloaded":
                print(f"[{completed}/{total}] {ts} {stream}: downloaded ({result.bytes_written / (1024 * 1024):.1f} MB)")
            elif result.status == "already_exists":
                print(f"[{completed}/{total}] {ts} {stream}: already exists")
            elif result.status == "missing":
                print(f"[{completed}/{total}] {ts} {stream}: MISSING")
            else:
                print(f"[{completed}/{total}] {ts} {stream}: FAILED: {result.error}")
    except KeyboardInterrupt:
        executor.shutdown(wait=False, cancel_futures=True)
        raise
    else:
        executor.shutdown(wait=True)

    print()
    print("Download summary:")
    print(f"  Downloaded:      {counts['downloaded']}")
    print(f"  Already existed: {counts['already_exists']}")
    print(f"  Missing:         {counts['missing']}")
    print(f"  Failed:          {counts['failed']}")
    return failed, missing, counts


# ---------------------------------------------------------------------------
# Spark ingestion
# ---------------------------------------------------------------------------
def run_spark_ingestion(start: date, end: date, streams: list):
    """Run spark/jobs/backfill_ingest.py inside the spark-master container
    (spark/jobs is mounted at /opt/jobs there)."""
    command = [
        "docker", "compose", "exec", "-T", "spark-master",
        "spark-submit", "--master", "spark://spark-master:7077",
        "/opt/jobs/backfill_ingest.py",
        "--start-date", start.strftime("%Y%m%d"),
        "--end-date", end.strftime("%Y%m%d"),
        "--streams", *streams,
    ]
    print()
    print("=" * 72)
    print(f"Starting Spark ingestion: {start} -> {end}")
    print("=" * 72)
    print("Command:", " ".join(command))

    result = subprocess.run(command, check=False, stdin=subprocess.DEVNULL)
    if result.returncode != 0:
        raise RuntimeError(
            f"Spark backfill exited with code {result.returncode} "
            "(look for Traceback or MISMATCH lines above)"
        )
    print()
    print(f"Spark ingestion completed successfully: {start} -> {end}")


# ---------------------------------------------------------------------------
# Validation
# ---------------------------------------------------------------------------
def validate_arguments(args):
    if args.start_date > args.end_date:
        raise SystemExit("--start-date cannot be after --end-date")
    if args.workers < 1:
        raise SystemExit("--workers must be >= 1")
    if args.retries < 1:
        raise SystemExit("--retries must be >= 1")
    if args.chunk_days < 1:
        raise SystemExit("--chunk-days must be >= 1")
    invalid = set(args.streams) - set(DEFAULT_STREAMS)
    if invalid:
        raise SystemExit(f"Invalid streams: {', '.join(sorted(invalid))}. Valid: {', '.join(DEFAULT_STREAMS)}")
    if args.download_only and args.ingest_only:
        raise SystemExit("--download-only and --ingest-only cannot be used together")

def _on_sigterm(signum, frame):
    raise KeyboardInterrupt


def run(args):
    start, end = args.start_date, args.end_date
    total_days = (end - start).days + 1

    print()
    print("=" * 72)
    print("GDELT BACKFILL")
    print("=" * 72)
    print(f"Date range:    {start} -> {end}")
    print(f"Days:          {total_days:,}")
    print(f"15-min slots:  {total_days * 96:,}")
    print(f"Files:         {total_days * 96 * len(args.streams):,}")
    print(f"Streams:       {', '.join(args.streams)}")
    print(f"Workers:       {args.workers}")
    print(f"Chunk size:    {args.chunk_days} days")
    print(f"S3 endpoint:   {S3_ENDPOINT}")
    print(f"S3 bucket:     {S3_BUCKET}")
    print("=" * 72)

    chunks = list(split_into_chunks(start, end, args.chunk_days))
    print(f"\nBackfill will run in {len(chunks)} chunk(s).")

    all_failed, all_missing, skipped_chunks = [], [], []
    totals = {"downloaded": 0, "already_exists": 0, "missing": 0, "failed": 0}

    # Each chunk is downloaded, then ingested. If chunk 27 fails, chunks 1-26 are done and a
    # re-run skips them (existing files are not re-downloaded; complete days are not re-ingested).
    for index, (chunk_start, chunk_end) in enumerate(chunks, start=1):
        print()
        print("#" * 72)
        print(f"CHUNK {index}/{len(chunks)}: {chunk_start} -> {chunk_end}")
        print("#" * 72)

        if not args.ingest_only:
            failed, missing, counts = download_chunk(
                chunk_start, chunk_end, args.streams, args.workers, args.retries
            )
            all_failed.extend(failed)
            all_missing.extend(missing)
            for k, v in counts.items():
                totals[k] += v

            # Never ingest a chunk with failed downloads (a 404 'missing' slot is fine).
            if failed:
                print()
                print(f"Chunk {chunk_start} -> {chunk_end} has {len(failed)} failed download(s).")
                print("Spark ingestion will NOT run for this chunk. Re-run the same command to retry.")
                skipped_chunks.append((chunk_start, chunk_end))
                continue

        if not args.download_only:
            try:
                run_spark_ingestion(chunk_start, chunk_end, args.streams)
            except Exception as exc:
                print()
                print(f"ERROR: Spark ingestion failed for {chunk_start} -> {chunk_end}")
                print(f"Reason: {exc}")
                print("Fix the cause and re-run the same command: downloaded files are kept "
                      "and days that are already complete are skipped.")
                raise SystemExit(1)

    # -------------------------------------------------------------------
    # Final summary
    # -------------------------------------------------------------------
    incomplete = bool(all_failed or skipped_chunks)
    print()
    print("=" * 72)
    print("BACKFILL INCOMPLETE" if incomplete else "BACKFILL COMPLETE")
    print("=" * 72)
    print(f"Range:   {start} -> {end}")
    print(f"Streams: {', '.join(args.streams)}")
    if not args.ingest_only:
        print(f"Downloads: {totals['downloaded']} new, {totals['already_exists']} already present, "
              f"{totals['missing']} missing (404), {totals['failed']} failed")

    if all_failed:
        print()
        print("Failed downloads:")
        for r in all_failed[:100]:
            print(f"  {r.timestamp} {r.stream}: {r.error}")
        if len(all_failed) > 100:
            print(f"  ... and {len(all_failed) - 100} more")
    if skipped_chunks:
        print()
        print("Chunks NOT ingested because of failed downloads:")
        for s, e in skipped_chunks:
            print(f"  {s} -> {e}")
    if all_missing:
        print()
        print(f"WARNING: {len(all_missing)} file(s) are not published by GDELT (404 after retries); "
              "their slots are simply absent:")
        for r in all_missing[:100]:
            print(f"  {r.timestamp} {r.stream}")
        if len(all_missing) > 100:
            print(f"  ... and {len(all_missing) - 100} more")
    if incomplete:
        print()
        print("Re-run the same command to retry; finished work is skipped.")

    if incomplete or (args.fail_on_missing and all_missing):
        sys.exit(1)


def main():
    parser = argparse.ArgumentParser(
        description="Download and backfill GDELT data into the Iceberg Bronze tables."
    )
    parser.add_argument("--start-date", required=True, type=parse_date, help="First date, YYYYMMDD")
    parser.add_argument("--end-date", required=True, type=parse_date, help="Last date, YYYYMMDD, inclusive")
    parser.add_argument("--streams", nargs="+", default=DEFAULT_STREAMS, choices=DEFAULT_STREAMS,
                        help="Streams to backfill. Default: events mentions gkg")
    parser.add_argument("--workers", type=int, default=DEFAULT_WORKERS,
                        help=f"Parallel download workers. Default: {DEFAULT_WORKERS}")
    parser.add_argument("--retries", type=int, default=DEFAULT_RETRIES,
                        help=f"Attempts per download. Default: {DEFAULT_RETRIES}")
    parser.add_argument("--chunk-days", type=int, default=DEFAULT_CHUNK_DAYS,
                        help=f"Days per download+Spark chunk. Default: {DEFAULT_CHUNK_DAYS}")
    parser.add_argument("--download-only", action="store_true",
                        help="Download files but do not run Spark ingestion.")
    parser.add_argument("--ingest-only", action="store_true",
                        help="Skip downloading; ingest the files already in MinIO.")
    parser.add_argument("--force", action="store_true",
                        help="Skip the 'another backfill is running' guards.")
    parser.add_argument("--fail-on-missing", action="store_true",
                        help="Exit 1 if GDELT never published some slots (default: warn only).")
    args = parser.parse_args()
    validate_arguments(args)

    signal.signal(signal.SIGTERM, _on_sigterm)

    lock = None
    if not args.force:
        lock = acquire_lock()      # keep a reference so it is held until the process exits
    preflight(args)

    try:
        run(args)
    except KeyboardInterrupt:
        print("\nInterrupted. Stopping this backfill's Spark application on the master...", flush=True)
        if not args.download_only:
            stopped = kill_backfill_apps()
            print(f"Targeted {stopped} Spark application(s).")
        print("Re-run the same command to resume; finished work is skipped.", flush=True)
        os._exit(130)   # do not wait for download threads that are sleeping in backoff


if __name__ == "__main__":
    main()

def parse_date(value: str) -> date:
    try:
        return datetime.strptime(value, "%Y%m%d").date()
    except ValueError:
        raise argparse.ArgumentTypeError(
            f"Invalid date '{value}'. Expected YYYYMMDD."
        )
def daterange(start: date, end: date):
    current = start
    while current <= end:
        yield current
        current += timedelta(days=1)
def timestamp_range(start: date, end: date):
    """
    Generate every 15-minute GDELT timestamp in the inclusive date range.
    96 timestamps per day.
    """
    current = datetime.combine(start, datetime.min.time())
    final = datetime.combine(end, datetime.max.time())
    while current <= final:
        yield current.strftime("%Y%m%d%H%M%S")
        current += timedelta(minutes=SLOT_MINUTES)
def split_into_chunks(
    start: date,
    end: date,
    chunk_days: int,
):
    """
    Split a potentially huge backfill into bounded date ranges.
    Example:
        2026-09-01 -> 2026-09-30
        chunk_days=7
    becomes:
        09-01 -> 09-07
        09-08 -> 09-14
        09-15 -> 09-21
        09-22 -> 09-28
        09-29 -> 09-30
    """
    current = start
    while current <= end:
        chunk_end = min(
            current + timedelta(days=chunk_days - 1),
            end,
        )
        yield current, chunk_end
        current = chunk_end + timedelta(days=1)

def object_exists(client, key: str) -> bool:
    """
    Check whether the landing object already exists.
    This is what makes the downloader resumable.
    """
    try:
        client.head_object(
            Bucket=S3_BUCKET,
            Key=key,
        )
        return True
    except ClientError as exc:
        error_code = exc.response.get("Error", {}).get("Code")
        if error_code in ("404", "NoSuchKey", "NotFound"):
            return False
        raise

# GDELT download
# ---------------------------------------------------------------------------
def download_one(
    timestamp: str,
    stream: str,
    max_retries: int,
) -> DownloadResult:
    key = f"{stream}/{timestamp}.csv"
    url = (
        f"{BASE_URL}/"
        f"{timestamp}."
        f"{STREAM_SUFFIX[stream]}"
    )
    client = s3_client()
    # ---------------------------------------------------------------
    # Resume support
    # ---------------------------------------------------------------
    try:
        if object_exists(client, key):
            return DownloadResult(
                timestamp=timestamp,
                stream=stream,
                status="already_exists",
            )
    except Exception as exc:
        return DownloadResult(
            timestamp=timestamp,
            stream=stream,
            status="failed",
            error=f"S3 HEAD failed: {exc}",
        )
    # ---------------------------------------------------------------
    # Download
    # ---------------------------------------------------------------
    for attempt in range(1, max_retries + 1):
        try:
            response = requests.get(
                url,
                timeout=REQUEST_TIMEOUT,
            )
            # -------------------------------------------------------
            # 404 is special for GDELT.
            #
            # A file can temporarily be unavailable because GDELT is
            # still processing it.
            # -------------------------------------------------------
            if response.status_code == 404:
                if attempt < max_retries:
                    wait = RETRY_BACKOFF[
                        min(attempt - 1, len(RETRY_BACKOFF) - 1)
                    ]
                    print(
                        f"[{timestamp}] "
                        f"{stream}: 404 "
                        f"(attempt {attempt}/{max_retries}), "
                        f"retrying in {wait}s"
                    )
                    time.sleep(wait)
                    continue
                return DownloadResult(
                    timestamp=timestamp,
                    stream=stream,
                    status="missing",
                    error="GDELT returned 404 after retries",
                )
            response.raise_for_status()
            # -------------------------------------------------------
            # Extract ZIP completely in memory for ONE file only.
            #
            # Because workers are bounded, memory usage is bounded.
            # -------------------------------------------------------
            with zipfile.ZipFile(
                io.BytesIO(response.content)
            ) as archive:
                members = archive.namelist()
                if not members:
                    raise RuntimeError(
                        "Downloaded ZIP contains no files"
                    )
                data = archive.read(members[0])
            # -------------------------------------------------------
            # Upload to MinIO
            # -------------------------------------------------------
            client.put_object(
                Bucket=S3_BUCKET,
                Key=key,
                Body=data,
            )
            return DownloadResult(
                timestamp=timestamp,
                stream=stream,
                status="downloaded",
                bytes_written=len(data),
            )
        except (
            requests.RequestException,
            zipfile.BadZipFile,
            ClientError,
        ) as exc:
            if attempt >= max_retries:
                return DownloadResult(
                    timestamp=timestamp,
                    stream=stream,
                    status="failed",
                    error=str(exc),
                )
            wait = RETRY_BACKOFF[
                min(attempt - 1, len(RETRY_BACKOFF) - 1)
            ]
            print(
                f"[{timestamp}] "
                f"{stream}: {type(exc).__name__} "
                f"(attempt {attempt}/{max_retries}), "
                f"retrying in {wait}s"
            )
            time.sleep(wait)
        except Exception as exc:
            return DownloadResult(
                timestamp=timestamp,
                stream=stream,
                status="failed",
                error=str(exc),
            )
    return DownloadResult(
        timestamp=timestamp,
        stream=stream,
        status="failed",
        error="Unknown download failure",
    )

# Parallel downloading

def download_chunk(
    start: date,
    end: date,
    streams: list[str],
    workers: int,
    retries: int,
):
    """
    Download one chunk using bounded parallelism.
    We submit only timestamp/stream identifiers to workers.
    The actual file contents never accumulate in the main process.
    """
    tasks = [
        (timestamp, stream)
        for timestamp in timestamp_range(start, end)
        for stream in streams
    ]
    total = len(tasks)
    print()
    print("=" * 72)
    print(
        f"Downloading {start} -> {end}"
    )
    print(
        f"Slots: {total // len(streams)}"
        f" | Streams: {', '.join(streams)}"
        f" | Files: {total}"
        f" | Workers: {workers}"
    )
    print("=" * 72)
    counts = {
        "downloaded": 0,
        "already_exists": 0,
        "missing": 0,
        "failed": 0,
    }
    failed_results = []
    completed = 0
    with concurrent.futures.ThreadPoolExecutor(
        max_workers=workers
    ) as executor:
        future_map = {
            executor.submit(
                download_one,
                timestamp,
                stream,
                retries,
            ): (timestamp, stream)
            for timestamp, stream in tasks
        }
        for future in concurrent.futures.as_completed(
            future_map
        ):
            timestamp, stream = future_map[future]
            try:
                result = future.result()
            except Exception as exc:
                result = DownloadResult(
                    timestamp=timestamp,
                    stream=stream,
                    status="failed",
                    error=str(exc),
                )
            completed += 1
            counts[result.status] += 1
            if result.status == "failed":
                failed_results.append(result)
            if result.status == "downloaded":
                size_mb = result.bytes_written / (1024 * 1024)
                print(
                    f"[{completed}/{total}] "
                    f"{timestamp} {stream}: "
                    f"downloaded ({size_mb:.1f} MB)"
                )
            elif result.status == "already_exists":
                print(
                    f"[{completed}/{total}] "
                    f"{timestamp} {stream}: "
                    f"already exists"
                )
            elif result.status == "missing":
                print(
                    f"[{completed}/{total}] "
                    f"{timestamp} {stream}: "
                    f"MISSING"
                )
            else:
                print(
                    f"[{completed}/{total}] "
                    f"{timestamp} {stream}: "
                    f"FAILED: {result.error}"
                )
    print()
    print("Download summary:")
    print(f"  Downloaded:      {counts['downloaded']}")
    print(f"  Already existed: {counts['already_exists']}")
    print(f"  Missing:         {counts['missing']}")
    print(f"  Failed:          {counts['failed']}")
    return failed_results

def run_spark_ingestion(
    start: date,
    end: date,
    streams: list[str],
):
    """
    Invoke the optimized Spark backfill job inside spark-master.
    scripts/ is not mounted into the Spark container, but
    spark/jobs/ is mounted at /opt/jobs, so we execute the existing
    /opt/jobs/backfill_ingest.py there.
    """
    command = [
        "docker",
        "compose",
        "exec",
        "-T",
        "spark-master",
        "spark-submit",
        "--master",
        "spark://spark-master:7077",
        "/opt/jobs/backfill_ingest.py",
        "--start-date",
        start.strftime("%Y%m%d"),
        "--end-date",
        end.strftime("%Y%m%d"),
        "--streams",
        *streams,
    ]
    print()
    print("=" * 72)
    print(
        f"Starting Spark ingestion: "
        f"{start} -> {end}"
    )
    print("=" * 72)
    print(
        "Command:",
        " ".join(command),
    )
    result = subprocess.run(
        command,
        check=False,
    )
    if result.returncode != 0:
        raise RuntimeError(
            f"Spark backfill failed with exit code "
            f"{result.returncode}"
        )
    print()
    print(
        f"Spark ingestion completed successfully: "
        f"{start} -> {end}"
    )

def validate_arguments(args):
    if args.start_date > args.end_date:
        raise SystemExit(
            "--start-date cannot be after --end-date"
        )
    if args.workers < 1:
        raise SystemExit(
            "--workers must be >= 1"
        )
    if args.retries < 1:
        raise SystemExit(
            "--retries must be >= 1"
        )
    if args.chunk_days < 1:
        raise SystemExit(
            "--chunk-days must be >= 1"
        )
    invalid = set(args.streams) - set(DEFAULT_STREAMS)
    if invalid:
        raise SystemExit(
            f"Invalid streams: {', '.join(sorted(invalid))}. "
            f"Valid streams: {', '.join(DEFAULT_STREAMS)}"
        )
    if args.download_only and args.ingest_only:
        raise SystemExit(
            "--download-only and --ingest-only "
            "cannot be used together"
        )

BASE_URL = "http://data.gdeltproject.org/gdeltv2"

    "events": "export.CSV.zip",
    "mentions": "mentions.CSV.zip",
    "gkg": "gkg.csv.zip",
}

def get_latest_timestamp() -> str:
    resp = requests.get(MANIFEST_URL, timeout=30)
    resp.raise_for_status()
    first_line = resp.text.strip().split("\n")[0]
    _, _, url = first_line.split()
    return url.rsplit("/", 1)[-1].split(".")[0]

def s3_client():
    return boto3.client(
        "s3",
        endpoint_url="http://localhost:9000",
        aws_access_key_id="admin",
        aws_secret_access_key="changeme123",
    )

def fetch_and_upload(timestamp: str, stream: str, client, max_retries=3, backoff_base=30):
    url = f"{BASE_URL}/{timestamp}.{STREAM_SUFFIX[stream]}"

    for attempt in range(1, max_retries + 1):
        resp = requests.get(url, timeout=60)

        if resp.status_code == 404:
            if attempt < max_retries:
                wait = backoff_base * attempt   # 30s, 60s, 90s
                print(f"  [{stream}] 404 on attempt {attempt}/{max_retries} — "
                      f"GDELT may still be processing this slot, retrying in {wait}s")
                time.sleep(wait)
                continue
            print(f"  [{stream}] SKIPPED after {max_retries} attempts — "
                  f"this slot genuinely has no {stream} file")
            return False

        resp.raise_for_status()
        with zipfile.ZipFile(io.BytesIO(resp.content)) as z:
            data = z.read(z.namelist()[0])

        key = f"{stream}/{timestamp}.csv"
        client.put_object(Bucket="gdelt-landing", Key=key, Body=data)
        print(f"  [{stream}] uploaded s3://gdelt-landing/{key} ({len(data):,} bytes)")
        return True

    return False

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--timestamp", help="YYYYMMDDHHMMSS, defaults to latest available")
    parser.add_argument("--streams", nargs="+", default=["events", "mentions", "gkg"])
    args = parser.parse_args()

    ts = args.timestamp or get_latest_timestamp()
    print(f"Batch timestamp: {ts}")

    client = s3_client()
    results = {stream: fetch_and_upload(ts, stream, client) for stream in args.streams}

    failed = [s for s, ok in results.items() if not ok]
    if failed:
        print(f"\nIncomplete batch — missing: {failed}")
    else:
        print(f"\nComplete batch: {ts}")

#!/usr/bin/env bash
# Test the tests: break the lakehouse on purpose, check that verify_all goes red on exactly the
# expected checks, undo the damage with Iceberg's rollback_to_snapshot, and check it is green again.
#
# SAFETY: before each fault the affected tables' current snapshot ids are written to
# <logdir>/pending_restore.txt, and an EXIT trap rolls them back even if this script fails midway.
# If the script is killed hard (kill -9, power loss), run the "--restore" command it printed.
#
# Usage: bash scripts/fault_injection_test.sh [START_YYYYMMDD] [END_YYYYMMDD]   (default: 20260901 20260930)
# Needs the stack up, no Spark application running, and Dagster OFF.
set -uo pipefail
cd "$(dirname "$0")/.."

START="${1:-20260901}"
END="${2:-20260930}"
LOGDIR="logs/fault_injection_$(date +%Y%m%d_%H%M%S)"
mkdir -p "$LOGDIR"
PENDING="$LOGDIR/pending_restore.txt"
: > "$PENDING"
FAILED_SCENARIOS=0

step() { printf '\n=== %s ===\n' "$1"; }

# < /dev/null: `docker compose exec -T` still reads stdin and would freeze a background job.
spark() {
  local script=$1; shift
  docker compose exec -T spark-master spark-submit --master spark://spark-master:7077 \
    "/opt/jobs/$script" "$@" < /dev/null
}

restore_pending() {
  [[ -s "$PENDING" ]] || return 0
  local spec
  spec=$(paste -sd, "$PENDING")
  echo "  restoring: $spec"
  if spark inject_fault.py --restore "$spec" > "$LOGDIR/restore.log" 2>&1; then
    : > "$PENDING"
    echo "  restored"
  else
    echo "  RESTORE FAILED. Undo it manually with:"
    echo "    docker compose exec -T spark-master spark-submit --master spark://spark-master:7077 /opt/jobs/inject_fault.py --restore \"$spec\" < /dev/null"
    return 1
  fi
}
trap restore_pending EXIT

# run_scenario <name> <required-failing-check-substrings, '|'-separated> <allowed-extra-failures, '|'-separated>
run_scenario() {
  local name=$1 required=$2 allowed=$3
  local dir="$LOGDIR/$name" ok=1 rc
  mkdir -p "$dir"
  step "Scenario: $name"

  spark inject_fault.py --scenario "$name" > "$dir/inject.log" 2>&1
  rc=$?
  # Record the snapshot ids BEFORE judging the result, so a half-applied fault is still undone.
  grep '^SNAP ' "$dir/inject.log" | awk '{print $2 "=" $3}' > "$PENDING" || true
  grep -E '^(INJECTED|APPENDED|DELETED|BLANKED)' "$dir/inject.log" || true
  if [[ $rc -ne 0 ]]; then
    echo "  FAIL: the fault could not be injected (see $dir/inject.log)"
    tail -n 8 "$dir/inject.log" | cut -c1-200
    restore_pending
    FAILED_SCENARIOS=$((FAILED_SCENARIOS + 1))
    return 1
  fi

  spark verify_all.py --start-date "$START" --end-date "$END" > "$dir/verify_after_fault.log" 2>&1
  rc=$?
  grep -E '^FAIL ' "$dir/verify_after_fault.log" | cut -c1-220 > "$dir/fails.txt" || true
  echo "  checks that failed after the fault:"
  if [[ -s "$dir/fails.txt" ]]; then sed 's/^/    /' "$dir/fails.txt"; else echo "    (none)"; fi

  if [[ $rc -eq 0 ]]; then
    echo "  FAIL: verify_all PASSED although the data was broken"
    ok=0
  fi

  local -a req alw
  IFS='|' read -ra req <<< "$required"
  IFS='|' read -ra alw <<< "$allowed"
  local r a line matched
  for r in "${req[@]}"; do
    grep -qF -- "$r" "$dir/fails.txt" || { echo "  FAIL: expected check did not fail: $r"; ok=0; }
  done
  while IFS= read -r line; do
    matched=0
    for r in "${req[@]}"; do [[ "$line" == *"$r"* ]] && matched=1; done
    for a in "${alw[@]}"; do [[ -n "$a" && "$line" == *"$a"* ]] && matched=1; done
    if [[ $matched -eq 0 ]]; then echo "  FAIL: unexpected extra failure: $line"; ok=0; fi
  done < "$dir/fails.txt"

  restore_pending || ok=0
  spark verify_all.py --start-date "$START" --end-date "$END" > "$dir/verify_after_restore.log" 2>&1
  if grep -qE '^SUMMARY: [0-9]+ passed, 0 failed' "$dir/verify_after_restore.log"; then
    echo "  PASS: verify_all is clean again after the rollback"
  else
    echo "  FAIL: verify_all is NOT clean after the rollback"
    grep -E '^(FAIL|SUMMARY)' "$dir/verify_after_restore.log" | cut -c1-200 || true
    ok=0
  fi

  if [[ $ok -eq 1 ]]; then
    echo "RESULT $name: PASS (the fault was detected by the right checks, and the rollback restored a clean state)"
  else
    echo "RESULT $name: FAIL"
    FAILED_SCENARIOS=$((FAILED_SCENARIOS + 1))
  fi
}

  exit 0
fi
echo "FAULT INJECTION TEST FAILED: $FAILED_SCENARIOS scenario(s). Logs in $LOGDIR"
exit 1

"""Bulk Bronze backfill: ONE read, ONE dedup, ONE append per day per stream.

Why this replaces the per-slot loop: dedup_and_append scans the existing keys of the
whole target table on every write. Per slot that is 288 scans a day, each bigger than
the last. Per day it is 3.

Behaviour:
  - A day is read with a glob (e.g. s3a://gdelt-landing/events/20260908*.csv). Slots
    GDELT never published simply have no file, so gaps need no special handling.
  - Completeness check: for each day/stream, the number of distinct 15-minute slots in
    Bronze must equal the number of slot files in the landing bucket. A day that already
    matches is skipped (so re-running a whole month is cheap); --no-skip forces a re-run,
    which is safe because the anti-join makes ingestion idempotent.
  - Real errors are NOT swallowed: any exception stops the run with a non-zero exit code.
    Re-running resumes, because finished days are skipped.
  - A day whose slot count still mismatches after ingest is reported as MISMATCH and the
    script exits 1 at the end.

Assumption: the same key never appears in two different slots of one day. (Per-slot
ingestion of Sept 1 found none.) If it did, both rows would be appended and
verify_bronze.py would flag the table.
"""
import argparse
import sys
import time
from datetime import datetime, timedelta

from pyspark.sql import SparkSession

from bronze_ingest import (
    ingest_events_batch,
    ingest_mentions_batch,
    ingest_gkg_batch,
)

LANDING = "s3a://gdelt-landing"

STREAMS = {
    "events": {"fn": ingest_events_batch, "table": "local.bronze.events", "ts_col": "date_added_ts"},
    "mentions": {"fn": ingest_mentions_batch, "table": "local.bronze.mentions", "ts_col": "mention_ts"},
    "gkg": {"fn": ingest_gkg_batch, "table": "local.bronze.gkg", "ts_col": "gkg_ts"},
}
VALID_STREAMS = list(STREAMS)


def date_range(start_date: str, end_date: str):
    """Every date from start_date to end_date inclusive, as YYYYMMDD strings."""
    start = datetime.strptime(start_date, "%Y%m%d")
    end = datetime.strptime(end_date, "%Y%m%d")
    if start > end:
        raise ValueError("--start-date cannot be after --end-date")
    current = start
    while current <= end:
        yield current.strftime("%Y%m%d")
        current += timedelta(days=1)


def landing_slots(spark, stream: str, day: str) -> int:
    """Number of slot files for this day present in the landing bucket."""
    path = spark._jvm.org.apache.hadoop.fs.Path(f"{LANDING}/{stream}/{day}*.csv")
    fs = path.getFileSystem(spark._jsc.hadoopConfiguration())
    matches = fs.globStatus(path)
    return 0 if matches is None else len(matches)


def bronze_slots(spark, stream: str, day: str) -> int:
    """Number of distinct 15-minute slot timestamps in Bronze for this day."""
    cfg = STREAMS[stream]
    ts = cfg["ts_col"]
    start = datetime.strptime(day, "%Y%m%d")
    end = start + timedelta(days=1)
    row = spark.sql(f"""
        SELECT COUNT(DISTINCT {ts}) AS n
        FROM {cfg['table']}
        WHERE {ts} >= TIMESTAMP '{start:%Y-%m-%d %H:%M:%S}'
          AND {ts} <  TIMESTAMP '{end:%Y-%m-%d %H:%M:%S}'
    """).first()
    return row["n"]


def main():
    parser = argparse.ArgumentParser(
        description="Bulk-backfill GDELT landing files into the Bronze Iceberg tables."
    )
    parser.add_argument("--date", help="Process a single date, YYYYMMDD.")
    parser.add_argument("--start-date", help="First date, YYYYMMDD.")
    parser.add_argument("--end-date", help="Last date, YYYYMMDD, inclusive.")
    parser.add_argument(
        "--streams", nargs="+", default=VALID_STREAMS, choices=VALID_STREAMS,
        help="Streams to ingest. Default: events mentions gkg",
    )
    parser.add_argument(
        "--no-skip", action="store_true",
        help="Re-run days that already look complete (still idempotent).",
    )
    args = parser.parse_args()

    if args.date and (args.start_date or args.end_date):
        parser.error("Use either --date OR --start-date/--end-date, not both.")
    if args.date:
        try:
            datetime.strptime(args.date, "%Y%m%d")
        except ValueError:
            parser.error(f"Invalid --date '{args.date}'. Expected YYYYMMDD.")
        dates = [args.date]
    else:
        if not args.start_date or not args.end_date:
            parser.error("Provide either --date or both --start-date and --end-date.")
        try:
            dates = list(date_range(args.start_date, args.end_date))
        except ValueError as e:
            parser.error(str(e))

    print("=" * 72)
    print("GDELT BULK BACKFILL (one append per day per stream)")
    print("=" * 72)
    print(f"Days:    {len(dates)} ({dates[0]} -> {dates[-1]})")
    print(f"Streams: {', '.join(args.streams)}")
    print(f"Skip complete days: {not args.no_skip}")
    print("=" * 72)

    spark = (
        SparkSession.builder.appName("gdelt-backfill-ingest")
        .config("spark.executor.memory", "4g")   # default is 1g; the worker has 6g
        .getOrCreate()
    )

    totals = {
        s: {"new": 0, "read": 0, "ingested": 0, "skipped": 0, "nofiles": 0, "mismatch": 0}
        for s in args.streams
    }
    started = time.perf_counter()

    try:
        for day_index, day in enumerate(dates, start=1):
            print()
            print(f"--- DAY {day_index}/{len(dates)}: {day} ---")
            for stream in args.streams:
                cfg = STREAMS[stream]
                files = landing_slots(spark, stream, day)
                if files == 0:
                    print(f"{day} {stream}: NO LANDING FILES (nothing to ingest)")
                    totals[stream]["nofiles"] += 1
                    continue

                have = bronze_slots(spark, stream, day)
                if have == files and not args.no_skip:
                    print(f"{day} {stream}: SKIP (already complete, {have}/{files} slots)")
                    totals[stream]["skipped"] += 1
                    continue

                t0 = time.perf_counter()
                new_rows, read_rows = cfg["fn"](
                    spark, f"{LANDING}/{stream}/{day}*.csv", cache=True
                )
                have_after = bronze_slots(spark, stream, day)
                status = "OK" if have_after == files else "MISMATCH"
                elapsed = time.perf_counter() - t0

                totals[stream]["new"] += new_rows
                totals[stream]["read"] += read_rows
                totals[stream]["ingested"] += 1
                if status == "MISMATCH":
                    totals[stream]["mismatch"] += 1

                print(
                    f"{day} {stream}: +{new_rows} new rows ({read_rows} read), "
                    f"slots {have_after}/{files} {status} ({elapsed:.0f}s)"
                )
    finally:
        spark.stop()

    print()
    print("=" * 72)
    print(f"BULK BACKFILL SUMMARY ({time.perf_counter() - started:.0f}s total)")
    print("=" * 72)
    for stream, t in totals.items():
        print(
            f"{stream}: {t['ingested']} days ingested, {t['skipped']} skipped, "
            f"{t['nofiles']} without files, {t['new']} new rows "

    if any(t["mismatch"] for t in totals.values()):

# spark/jobs/bronze_ingest.py
import argparse
from functools import reduce

from pyspark import StorageLevel
from pyspark.sql import SparkSession, functions as F
from pyspark.sql.types import StructType
from schemas import EVENTS_COLUMNS, MENTIONS_COLUMNS, GKG_COLUMNS
from utils import parse_gdelt_ts

# A source mention row is only identified by event + article + time + sentence +
# the three char offsets + Confidence. (GLOBALEVENTID, MentionIdentifier, mention_ts)
# alone collides: the same article can mention an event in several sentences, GDELT
# also emits rows that differ only in Actor1/Actor2 char offsets, and (observed on
# 2026-09-01) rows that differ only in Confidence (10 vs 20).
MENTIONS_KEY = [
    "GLOBALEVENTID", "MentionIdentifier", "mention_ts",
    "SentenceID", "Actor1CharOffset", "Actor2CharOffset", "ActionCharOffset",
    "Confidence",
]


def dedup_and_append(spark, batch_df, target_table: str, key_cols: list,
                     drop_exact_dupes: bool = False, cache_new_rows: bool = False):
    """Insert-only idempotency via an explicit left-anti join, since Iceberg's
    MERGE INTO insert-only optimization was observed to append duplicates
    unconditionally rather than checking existing keys.

    - The anti-join is null-safe (<=>), so a row with a NULL in a key column is
      still recognised on re-ingest instead of being appended every time.
    - drop_exact_dupes removes rows that are identical in EVERY column within the
      batch (the anti-join only dedups against rows already in the table).
    - cache_new_rows persists the anti-join result so the expensive scan of the
      existing keys runs once instead of twice (count, then write). Used by the
      bulk backfill; the live per-slot pipeline leaves it off.
    - total_count is the raw source row count, before any dedup.
    - Rows that share a key but differ in content are NOT dropped (the verify step
      flags them); only exact copies and rows already in the table are removed.
    """
    total_count = batch_df.count()

    if drop_exact_dupes:
        batch_df = batch_df.dropDuplicates()

    existing_keys = spark.table(target_table).select(*key_cols).distinct()
    match = reduce(
        lambda a, b: a & b,
        [F.col(f"b.{c}").eqNullSafe(F.col(f"e.{c}")) for c in key_cols],
    )
    new_rows = batch_df.alias("b").join(existing_keys.alias("e"), match, "left_anti")

    if cache_new_rows:
        new_rows = new_rows.persist(StorageLevel.MEMORY_AND_DISK)
    try:
        new_count = new_rows.count()
        if new_count > 0:
            new_rows.writeTo(target_table).append()
    finally:
        if cache_new_rows:
            new_rows.unpersist()
    return new_count, total_count


def ingest_events_batch(spark, path: str, cache: bool = False):
    print(f"Reading events from: {path}")
    target_table = "local.bronze.events"
    target_schema = spark.table(target_table).schema

    # Map exact CSV column order to Iceberg data types
    csv_schema = StructType([target_schema[col] for col in EVENTS_COLUMNS if col in target_schema.names])

    raw_df = spark.read \
        .option("sep", "\t") \
        .option("nullValue", "") \
        .schema(csv_schema) \
        .csv(path)

    batch_df = raw_df.withColumn("date_added_ts", parse_gdelt_ts("DATEADDED"))
    return dedup_and_append(spark, batch_df, target_table, ["GLOBALEVENTID"],
                            cache_new_rows=cache)


def ingest_mentions_batch(spark, path: str, cache: bool = False):
    print(f"Reading mentions from: {path}")
    target_table = "local.bronze.mentions"
    target_schema = spark.table(target_table).schema

    csv_schema = StructType([target_schema[col] for col in MENTIONS_COLUMNS if col in target_schema.names])

    raw_df = spark.read \
        .option("sep", "\t") \
        .option("nullValue", "") \
        .schema(csv_schema) \
        .csv(path)

    batch_df = raw_df.withColumn("mention_ts", parse_gdelt_ts("MentionTimeDate"))
    return dedup_and_append(
        spark, batch_df, target_table, MENTIONS_KEY,
        drop_exact_dupes=True, cache_new_rows=cache,
    )


def ingest_gkg_batch(spark, path: str, cache: bool = False):
    print(f"Reading gkg from: {path}")
    target_table = "local.bronze.gkg"
    target_schema = spark.table(target_table).schema

    csv_schema = StructType([target_schema[col] for col in GKG_COLUMNS if col in target_schema.names])

    raw_df = spark.read \
        .option("sep", "\t") \
        .option("nullValue", "") \
        .schema(csv_schema) \
        .csv(path)

    batch_df = raw_df.withColumn("gkg_ts", parse_gdelt_ts("DATE"))
    return dedup_and_append(spark, batch_df, target_table, ["GKGRECORDID"],
                            cache_new_rows=cache)


STREAM_FUNCS = {
    "events": ingest_events_batch,
    "mentions": ingest_mentions_batch,
    "gkg": ingest_gkg_batch,
}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--stream", required=True, choices=STREAM_FUNCS.keys())
    parser.add_argument("--path", required=True)
    args = parser.parse_args()

    spark = SparkSession.builder.appName(f"bronze-ingest-{args.stream}").getOrCreate()
    new_count, total_count = STREAM_FUNCS[args.stream](spark, args.path)
    print(f"{args.stream}: {new_count} new rows written, {total_count} rows in source batch "
          f"({total_count - new_count} already existed or were exact in-batch copies)")
    spark.stop()


if __name__ == "__main__":
    main()

"""Read-only: after widening the mentions key, show what still collides.

For every group of rows sharing the full MENTIONS_KEY it reports which NON-key
columns differ between the copies, then prints the colliding rows themselves.
"""
from pyspark.sql import SparkSession, functions as F
from bronze_ingest import MENTIONS_KEY


def main():
    spark = SparkSession.builder.appName("diagnose-key-collisions").getOrCreate()

    m = spark.table("local.bronze.mentions")
    other_cols = [c for c in m.columns if c not in MENTIONS_KEY]

    def nd(c):
        # distinct count that treats NULL as a value (plain countDistinct ignores NULLs)
        return F.countDistinct(F.coalesce(F.col(c).cast("string"), F.lit("<NULL>")))

    per_key = m.groupBy(*MENTIONS_KEY).agg(
        F.count("*").alias("n"),
        *[nd(c).alias(f"d_{c}") for c in other_cols],
    )
    dups = per_key.filter("n > 1").cache()

    print("=== COLLISION SUMMARY ===")
    print("(each column = number of colliding groups where that column differs between the copies)")
    dups.agg(
        F.count("*").alias("dup_groups"),
        F.sum(F.col("n") - 1).alias("extra_rows"),
        *[F.sum(F.when(F.col(f"d_{c}") > 1, 1).otherwise(0)).alias(c) for c in other_cols],
    ).show(vertical=True, truncate=False)

    print("=== COLLISION ROWS ===")
    (m.join(dups.select(*MENTIONS_KEY), on=MENTIONS_KEY, how="inner")
       .select("GLOBALEVENTID", "SentenceID", "Actor1CharOffset", "Actor2CharOffset",
               "ActionCharOffset", *other_cols)
       .orderBy("GLOBALEVENTID", "SentenceID")
       .show(40, truncate=40))

    spark.stop()


if __name__ == "__main__":
    main()

"""Read-only evidence for the Silver mention-grain decision.

from utils import normalize_url

CHECK_COLS = ["MentionSourceName", "MentionDocTone", "MentionDocLen",
              "Confidence", "SentenceID", "InRawText"]


def main():
    spark = (
        SparkSession.builder.appName("diagnose-mention-grain")
        .config("spark.executor.memory", "4g")
        .getOrCreate()
    )

        spark.table("local.bronze.events")
        .filter(
            (F.col("ActionGeo_CountryCode") == "IN")

        .select("GLOBALEVENTID")
        .distinct()

"""Read-only: explain why bronze.mentions has more rows than distinct
(GLOBALEVENTID, MentionIdentifier, mention_ts) keys.

For each duplicated key it reports:
  - n_files: how many distinct Iceberg data files the copies live in
    (each backfill batch writes one file, so >1 means cross-batch, 1 means within-batch)
  - n_sentence_ids: whether the copies differ by SentenceID
  - n_distinct_rows: whether the copies are fully identical (1) or differ in some column
"""
from pyspark.sql import SparkSession, functions as F

KEY = ["GLOBALEVENTID", "MentionIdentifier", "mention_ts"]


def main():
    spark = SparkSession.builder.appName("diagnose-mention-dups").getOrCreate()

    m = spark.sql("SELECT *, _file AS src_file FROM local.bronze.mentions")
    data_cols = [c for c in m.columns if c != "src_file"]

    per_key = m.groupBy(*KEY).agg(
        F.count("*").alias("n"),
        F.countDistinct("src_file").alias("n_files"),
        F.countDistinct("SentenceID").alias("n_sentence_ids"),
        F.countDistinct(F.sha2(F.to_json(F.struct(*data_cols)), 256)).alias("n_distinct_rows"),
    )
    dups = per_key.filter("n > 1").cache()

    summary = dups.agg(
        F.count("*").alias("dup_groups"),
        F.sum(F.col("n") - 1).alias("extra_rows"),
        F.sum(F.when(F.col("n_files") > 1, 1).otherwise(0)).alias("groups_across_files"),
        F.sum(F.when(F.col("n_files") == 1, 1).otherwise(0)).alias("groups_within_one_file"),
        F.sum(F.when(F.col("n_sentence_ids") > 1, 1).otherwise(0)).alias("groups_differing_sentenceid"),
        F.sum(F.when(F.col("n_distinct_rows") == 1, 1).otherwise(0)).alias("groups_fully_identical"),
    )
    print("=== DUP SUMMARY ===")
    summary.show(truncate=False)

    print("=== SAMPLE DUPLICATED KEYS (5 groups) ===")
    sample_keys = dups.select(*KEY).limit(5)
    (m.join(sample_keys, on=KEY, how="inner")
       .select("GLOBALEVENTID", "MentionIdentifier", "mention_ts", "SentenceID",
               "Actor1CharOffset", "Actor2CharOffset", "ActionCharOffset",
               "InRawText", "Confidence", "MentionDocLen",
               F.regexp_extract("src_file", r"([^/]+)$", 1).alias("src_file"))
       .orderBy("GLOBALEVENTID", "MentionIdentifier", "SentenceID")
       .show(30, truncate=60))

    spark.stop()


if __name__ == "__main__":
    main()

from pyspark.sql import SparkSession, functions as F
from utils import partner_country

# The two daily tables are full recomputes, so they are replaced in one commit with
# overwrite(F.lit(True)). overwritePartitions() only replaces partitions that appear in the
# new data, so a month that no longer has any rows would keep its old rows forever.


def compute_bilateral_sentiment(events):
    """Silver events -> one row per (event_date, partner country). Only events where India
    is exactly one of the two actors count; the other actor is the partner."""
    return (
        events
        .withColumn(
            "partner_country",
            partner_country(F.col("Actor1CountryCode"), F.col("Actor2CountryCode")),
        )
        .filter(F.col("partner_country").isNotNull() & (F.col("partner_country") != ""))
        .groupBy("event_date", "partner_country")
        .agg(
            F.count("*").alias("event_count"),
            F.avg("GoldsteinScale").alias("avg_goldstein"),
            F.avg("AvgTone").alias("avg_tone"),
        )
    )


def compute_state_conflict(events):
    """Silver events -> one row per (event_date, Indian state, severity). Only events that
    happened in India count."""
    return (
        events
        .filter(
            (F.col("ActionGeo_CountryCode") == "IN") &
            F.col("ActionGeo_ADM1Code").isNotNull()
        )
        .groupBy("event_date", "ActionGeo_ADM1Code", "conflict_severity")
        .agg(
            F.count("*").alias("event_count"),
            F.avg("GoldsteinScale").alias("avg_goldstein"),
        )
        .withColumnRenamed("ActionGeo_ADM1Code", "adm1_code")
    )


def compute_mention_counts(mentions):
    """Silver mentions -> one row per event with its FULL current totals."""
    return (
        mentions
        .groupBy("GLOBALEVENTID")
        .agg(
            F.count("*").alias("total_mentions"),
            F.countDistinct("MentionSourceName").alias("total_sources"),
            F.avg("MentionDocTone").alias("running_avg_tone"),
        )
    )


def build_bilateral_sentiment(spark):
    gold = compute_bilateral_sentiment(spark.table("local.silver.events_india"))
    row_count = gold.count()
    gold.writeTo("local.gold.india_bilateral_sentiment_daily").overwrite(F.lit(True))
    return row_count


def build_state_conflict(spark):
    gold = compute_state_conflict(spark.table("local.silver.events_india"))
    row_count = gold.count()
    gold.writeTo("local.gold.india_state_conflict_daily").overwrite(F.lit(True))
    return row_count


def merge_mention_counts(spark):
    """Upsert against the FULL current total for each event -- not an additive
    delta. Silver is a full recompute, not incremental, so this MERGE must replace
    each event's totals rather than add to them, or re-running the pipeline would
    double-count on every run."""
    agg = compute_mention_counts(spark.table("local.silver.mentions_enriched"))
    agg.createOrReplaceTempView("mention_agg_source")

    spark.sql("""
        MERGE INTO local.gold.india_event_mention_counts t
        USING mention_agg_source s
        ON t.GLOBALEVENTID = s.GLOBALEVENTID
        WHEN MATCHED THEN UPDATE SET
            t.total_mentions    = s.total_mentions,
            t.total_sources     = s.total_sources,
            t.running_avg_tone  = s.running_avg_tone,
            t.last_updated_ts   = current_timestamp()
        WHEN NOT MATCHED THEN INSERT (
            GLOBALEVENTID, total_mentions, total_sources, running_avg_tone, last_updated_ts
        ) VALUES (
            s.GLOBALEVENTID, s.total_mentions, s.total_sources, s.running_avg_tone, current_timestamp()
        )
    """)

    return agg.count()


def main():
    spark = SparkSession.builder.appName("gold-aggregate").getOrCreate()

    b = build_bilateral_sentiment(spark)
    print(f"gold.india_bilateral_sentiment_daily: {b} rows written")

    s = build_state_conflict(spark)
    print(f"gold.india_state_conflict_daily: {s} rows written")

    m = merge_mention_counts(spark)
    print(f"gold.india_event_mention_counts: merged against {m} distinct events")

    spark.stop()


if __name__ == "__main__":
    main()

"""Deliberately damage one table so the acceptance checks can be tested, and undo it exactly.

Every scenario first prints 'SNAP <table> <snapshot_id>' for each table it will touch (the
table's CURRENT snapshot, taken from the 'main' ref), then makes the change. --restore rolls
each table back to its snapshot with Iceberg's rollback_to_snapshot procedure.

    --scenario dup_event | missing_slot | gold_drift | gkg_stall
    --restore  bronze.events=123,silver.mentions_enriched=456

Never run this while the live pipeline is writing to the same tables.
"""
import argparse

from pyspark.sql import SparkSession

SLOT_TO_DELETE = "2026-09-10 12:00:00"


def snap(spark, table):
    row = spark.sql(f"SELECT snapshot_id FROM local.{table}.refs WHERE name = 'main'").first()
    if row is None:
        raise RuntimeError(f"no 'main' snapshot found for {table}")
    print(f"SNAP {table} {row['snapshot_id']}", flush=True)


def dup_event(spark):
    snap(spark, "bronze.events")
    spark.table("local.bronze.events").limit(1).writeTo("local.bronze.events").append()
    print("APPENDED 1 duplicate row to bronze.events", flush=True)


def missing_slot(spark):
    snap(spark, "bronze.mentions")
    n = spark.table("local.bronze.mentions").filter(f"mention_ts = TIMESTAMP '{SLOT_TO_DELETE}'").count()
    spark.sql(f"DELETE FROM local.bronze.mentions WHERE mention_ts = TIMESTAMP '{SLOT_TO_DELETE}'")
    print(f"DELETED {n} rows (slot {SLOT_TO_DELETE}) from bronze.mentions", flush=True)


def gold_drift(spark):
    snap(spark, "gold.india_bilateral_sentiment_daily")
    spark.sql("""
        INSERT INTO local.gold.india_bilateral_sentiment_daily
        VALUES (DATE '2026-09-15', 'ZZZ', CAST(5 AS BIGINT), CAST(0.0 AS DOUBLE), CAST(0.0 AS DOUBLE))
    """)
    print("APPENDED 1 bogus row to gold.india_bilateral_sentiment_daily", flush=True)


def gkg_stall(spark):
    snap(spark, "silver.mentions_enriched")
    spark.sql("""
        UPDATE local.silver.mentions_enriched
        SET V2Themes = NULL, V2Tone = NULL
        WHERE V2Themes IS NOT NULL
    """)
    print("BLANKED the GKG columns of silver.mentions_enriched", flush=True)


SCENARIOS = {
    "dup_event": dup_event,
    "missing_slot": missing_slot,
    "gold_drift": gold_drift,
    "gkg_stall": gkg_stall,
}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--scenario", choices=list(SCENARIOS))
    parser.add_argument("--restore", help="comma-separated table=snapshot_id pairs")
    args = parser.parse_args()
    if bool(args.scenario) == bool(args.restore):
        parser.error("give exactly one of --scenario or --restore")

    spark = SparkSession.builder.appName("inject-fault").getOrCreate()
    try:
        if args.scenario:
            SCENARIOS[args.scenario](spark)
            print(f"INJECTED {args.scenario}", flush=True)
        else:
            for pair in args.restore.split(","):
                table, snapshot_id = pair.split("=")
                spark.sql(f"CALL local.system.rollback_to_snapshot('{table}', {int(snapshot_id)})")
                print(f"RESTORED {table} to snapshot {snapshot_id}", flush=True)
    finally:
        spark.stop()


if __name__ == "__main__":
    main()

"""Drop every table and namespace in the lakehouse so Iceberg is back to zero.

DROP TABLE ... PURGE removes the table registration in the REST catalog AND
deletes its data/metadata files from MinIO. Namespaces are dropped last.
Recreate everything afterwards with bootstrap_bronze/silver/gold.py.
"""
from pyspark.sql import SparkSession

NAMESPACES = ["bronze", "silver", "gold", "scratch"]


def main():
    spark = SparkSession.builder.appName("reset-lakehouse").getOrCreate()

    existing = {row[0] for row in spark.sql("SHOW NAMESPACES IN local").collect()}
    print(f"Namespaces before reset: {sorted(existing)}")

    for ns in NAMESPACES:
        if ns not in existing:
            print(f"[skip] local.{ns} does not exist")
            continue

        for t in spark.sql(f"SHOW TABLES IN local.{ns}").collect():
            fqn = f"local.{ns}.{t.tableName}"
            spark.sql(f"DROP TABLE {fqn} PURGE")
            print(f"[dropped] {fqn}")

        spark.sql(f"DROP NAMESPACE local.{ns}")
        print(f"[dropped] namespace local.{ns}")

    print("\nNamespaces after reset:")
    spark.sql("SHOW NAMESPACES IN local").show(truncate=False)
    spark.stop()


if __name__ == "__main__":
    main()

from pyspark.sql import SparkSession, functions as F
from utils import normalize_url, conflict_severity


def build_events_india(bronze_events):
    """Bronze events -> Silver India events. Pure DataFrame logic (no table access), so it
    can be unit-tested on small fixtures."""
    return (
        bronze_events
        .withColumn("event_date", F.to_date("SQLDATE", "yyyyMMdd"))
        .filter(
            (F.col("ActionGeo_CountryCode") == "IN") |
            (F.col("Actor1CountryCode") == "IND") |
            (F.col("Actor2CountryCode") == "IND")
        )
        .filter(F.col("SOURCEURL").isNotNull() & F.col("GLOBALEVENTID").isNotNull())
        .withColumn("conflict_severity", conflict_severity(F.col("QuadClass")))
        .select(
            "GLOBALEVENTID", "event_date", "Actor1CountryCode", "Actor1Name",
            "Actor2CountryCode", "Actor2Name", "EventRootCode", "QuadClass",
            "conflict_severity", "GoldsteinScale", "AvgTone",
            "ActionGeo_CountryCode", "ActionGeo_ADM1Code", "ActionGeo_Lat", "ActionGeo_Long",
            "SOURCEURL"
        )
        .dropDuplicates(["GLOBALEVENTID"])  # defensive; Bronze should already guarantee this
    )


def transform_events_india(spark):
    silver_events = build_events_india(spark.table("local.bronze.events"))
    row_count = silver_events.count()
    silver_events.writeTo("local.silver.events_india").overwritePartitions()
    return row_count


def build_mentions_enriched(bronze_mentions, bronze_gkg, silver_events):
    """One row per (GLOBALEVENTID, MentionIdentifier), i.e. per article that mentions an
    India-relevant event.

    Design, each step backed by what we measured on September's Bronze data:
    1. Scope to India events FIRST (small, broadcast), then collapse to the article grain.
       Bronze has 2 rows for ~2% of (event, article) pairs. MentionSourceName,
       MentionDocTone and MentionDocLen were identical in all of them, so they are safe to
       collapse (min). Confidence differed in some pairs, so the highest value is kept.
       SentenceID / char offsets / InRawText are per-sentence detail that Silver does not carry.
    2. Pull only the GKG rows these articles can match, then keep ONE GKG row per
       normalized URL (the latest gkg_ts, ties broken by GKGRECORDID). About 1.6% of GKG rows
       share a normalized URL with another row; without this a LEFT JOIN would multiply mentions.
       This tie-break is a deterministic choice, not a verified 'best' row.
    3. LEFT JOIN, so mentions without a GKG record (e.g. during a GKG stall) are kept.
    """
    silver_events = silver_events.select("GLOBALEVENTID", "event_date")

    pairs = (
        bronze_mentions.alias("m")
        .join(
            F.broadcast(silver_events.alias("e")),
            F.col("m.GLOBALEVENTID") == F.col("e.GLOBALEVENTID"),
            "inner",   # scope to India-relevant events only
        )
        .groupBy(
            F.col("m.GLOBALEVENTID").alias("GLOBALEVENTID"),
            F.col("m.MentionIdentifier").alias("MentionIdentifier"),
        )
        .agg(
            F.min("e.event_date").alias("event_date"),
            F.min("m.MentionSourceName").alias("MentionSourceName"),
            F.min("m.MentionDocTone").alias("MentionDocTone"),
            F.max("m.Confidence").alias("Confidence"),
        )
        .withColumn("norm_url", normalize_url(F.col("MentionIdentifier")))
    )

    gkg_one_per_url = (
        bronze_gkg
        .filter(F.col("DocumentIdentifier").isNotNull())   # also drops malformed fragment rows
        .withColumn("norm_url", normalize_url(F.col("DocumentIdentifier")))
        .join(pairs.select("norm_url").distinct(), "norm_url", "left_semi")
        .groupBy("norm_url")
        .agg(F.max(F.struct("gkg_ts", "GKGRECORDID", "V2Themes", "V2Tone")).alias("g"))
        .select(
            "norm_url",
            F.col("g.V2Themes").alias("V2Themes"),
            F.col("g.V2Tone").alias("V2Tone"),
        )
    )

    return (
        pairs.join(gkg_one_per_url, "norm_url", "left")
        .select(
            "GLOBALEVENTID", "event_date", "MentionSourceName", "MentionIdentifier",
            "MentionDocTone", "Confidence", "V2Themes", "V2Tone",
        )
    )


def transform_mentions_enriched(spark):
    silver_mentions = build_mentions_enriched(
        spark.table("local.bronze.mentions"),
        spark.table("local.bronze.gkg"),
        spark.table("local.silver.events_india"),
    )
    row_count = silver_mentions.count()
    silver_mentions.writeTo("local.silver.mentions_enriched").overwritePartitions()
    return row_count


def main():
    spark = (
        SparkSession.builder.appName("silver-transform")
        .config("spark.executor.memory", "4g")   # default is 1g; the worker has 6g
        .getOrCreate()
    )
    events_count = transform_events_india(spark)
    print(f"silver.events_india: {events_count} rows written")
    mentions_count = transform_mentions_enriched(spark)
    print(f"silver.mentions_enriched: {mentions_count} rows written")
    spark.stop()


if __name__ == "__main__":
    main()

"""Read-only: print one fingerprint line per lakehouse table:  FP <table> <row_count> <content_hash>

Bronze tables are append-only, so only the row count is fingerprinted (hash shown as '-').
Silver and Gold are recomputed, so a content hash is added: an order-independent sum of a
per-row xxhash64. Two details keep the comparison fair:
  * doubles are rounded to 6 decimals first, so harmless floating-point summation-order
    noise in averages is not reported as a difference
  * gold.india_event_mention_counts.last_updated_ts is excluded; it changes on every MERGE by design
"""
from pyspark.sql import SparkSession, functions as F
from pyspark.sql.types import DoubleType

BRONZE = ["bronze.events", "bronze.mentions", "bronze.gkg"]
RECOMPUTED = {
    "silver.events_india": [],
    "silver.mentions_enriched": [],
    "gold.india_bilateral_sentiment_daily": [],
    "gold.india_state_conflict_daily": [],
    "gold.india_event_mention_counts": ["last_updated_ts"],
}


def content_fingerprint(df, exclude):
    cols = []
    for field in df.schema.fields:
        if field.name in exclude:
            continue
        c = F.col(field.name)
        if isinstance(field.dataType, DoubleType):
            c = F.round(c, 6)
        cols.append(c)
    row = df.agg(F.count("*").alias("n"), F.sum(F.xxhash64(*cols)).alias("h")).first()
    return row["n"], (row["h"] if row["h"] is not None else 0)


def main():
    spark = (
        SparkSession.builder.appName("table-fingerprint")
        .config("spark.executor.memory", "4g")
        .getOrCreate()
    )

    for table in BRONZE:
        n = spark.table(f"local.{table}").count()
        print(f"FP {table} {n} -", flush=True)

    for table, exclude in RECOMPUTED.items():
        n, h = content_fingerprint(spark.table(f"local.{table}"), exclude)
        print(f"FP {table} {n} {h}", flush=True)

    spark.stop()


if __name__ == "__main__":
    main()

"""Read-only: file counts / sizes / partitions per lakehouse table, plus a first look at
Silver event dates and the Gold bilateral table. Feeds the compaction decision.

from pyspark.sql import SparkSession

TABLES = [
    "bronze.events", "bronze.mentions", "bronze.gkg",
    "silver.events_india", "silver.mentions_enriched",
    "gold.india_bilateral_sentiment_daily", "gold.india_state_conflict_daily",
    "gold.india_event_mention_counts",
]


def main():
    spark = SparkSession.builder.appName("table-health").getOrCreate()

    print("=== TABLE HEALTH ===")
    for t in TABLES:
        f = spark.sql(
            f"SELECT COUNT(*) AS n, SUM(file_size_in_bytes) AS bytes, SUM(record_count) AS recs "
            f"FROM local.{t}.files"
        ).first()
        parts = spark.sql(f"SELECT COUNT(*) AS n FROM local.{t}.partitions").first()["n"]
        snaps = spark.sql(f"SELECT COUNT(*) AS n FROM local.{t}.snapshots").first()["n"]
        n_files = f["n"] or 0
        size_mb = (f["bytes"] or 0) / 1e6
        avg_mb = size_mb / n_files if n_files else 0.0
        print(
            f"{t:40s} files={n_files:6d} partitions={parts:5d} "
            f"size_mb={size_mb:9.1f} avg_file_mb={avg_mb:7.2f} rows={f['recs'] or 0:9d} snapshots={snaps:4d}"
        )

    print("=== SILVER EVENT DATES ===")
    spark.sql("""
        SELECT COUNT(*)                                                 AS row_count,
               MIN(event_date)                                          AS min_date,
               MAX(event_date)                                          AS max_date,

        FROM local.silver.events_india

        SELECT partner_country, SUM(event_count) AS events, COUNT(*) AS day_rows
        FROM local.gold.india_bilateral_sentiment_daily

"""Standalone check for MERGE INTO ... WHEN MATCHED THEN UPDATE, run once to
build confidence before Gold's incremental counters depend on it. Uses a
disposable scratch table -- doesn't touch bronze/silver/gold."""
from pyspark.sql import SparkSession

def main():
    spark = SparkSession.builder.appName("test-merge-update").getOrCreate()

    spark.sql("CREATE NAMESPACE IF NOT EXISTS local.scratch")
    spark.sql("DROP TABLE IF EXISTS local.scratch.merge_test")
    spark.sql("""
        CREATE TABLE local.scratch.merge_test (id BIGINT, val STRING)
        USING iceberg TBLPROPERTIES ('format-version'='2')
    """)
    spark.sql("INSERT INTO local.scratch.merge_test VALUES (1, 'original'), (2, 'original')")

    spark.createDataFrame([(1, "updated"), (3, "new")], ["id", "val"]) \
        .createOrReplaceTempView("merge_source")

    spark.sql("""
        MERGE INTO local.scratch.merge_test t
        USING merge_source s
        ON t.id = s.id
        WHEN MATCHED THEN UPDATE SET t.val = s.val
        WHEN NOT MATCHED THEN INSERT *
    """)

    rows = spark.sql("SELECT * FROM local.scratch.merge_test ORDER BY id").collect()
    actual = {row.id: row.val for row in rows}
    expected = {1: "updated", 2: "original", 3: "new"}

    print("Result after MERGE:", actual)
    print("PASS" if actual == expected else f"FAIL -- expected {expected}, got {actual}")

    spark.stop()

if __name__ == "__main__":
    main()

from pyspark.sql import functions as F

# Query parameters that only track where a click came from. Anything else in the query
# string (id=, p=, story=, ...) can identify the article itself and must be kept.
TRACKING_PARAMS = (
    r"utm_[a-z0-9_]*|fbclid|gclid|dclid|msclkid|mc_cid|mc_eid|igshid|ocid|ncid|cmpid"
)


def normalize_url(col):
    """Lowercase, trim, drop the #fragment and tracking parameters -- for the
    Mentions<->GKG soft join.

    Unlike the previous version this keeps the rest of the query string, so that
    'site.com/story.php?id=1' and 'site.com/story.php?id=2' stay different articles."""
    s = F.lower(F.trim(col))
    s = F.regexp_replace(s, r"#.*$", "")
    s = F.regexp_replace(s, r"[?&](" + TRACKING_PARAMS + r")(=[^&]*)?", "")
    # if the first parameter was removed, the next one starts with '&' instead of '?'
    s = F.regexp_replace(s, r"^([^?]*)&", "$1?")
    return s


def conflict_severity(quad_class):
    """GDELT QuadClass -> label. Unknown or NULL QuadClass stays NULL (it used to be
    silently labelled 'Verbal Cooperation')."""
    return (
        F.when(quad_class == 4, "Material Conflict")
         .when(quad_class == 3, "Verbal Conflict")
         .when(quad_class == 2, "Material Cooperation")
         .when(quad_class == 1, "Verbal Cooperation")
    )


def partner_country(actor1_country, actor2_country):
    """The country India interacts with in an event, or NULL when India is not exactly
    one of the two actors (India-India, no India actor, or the other actor is missing).
    Events that merely happened in India no longer get Actor1 as a 'partner'."""
    return (
        F.when((actor1_country == "IND") & (actor2_country != "IND"), actor2_country)
         .when((actor2_country == "IND") & (actor1_country != "IND"), actor1_country)
    )

def parse_gdelt_ts(col, fmt="yyyyMMddHHmmss"):
    return F.to_timestamp(col, fmt)

def extract_batch_id(file_path: str) -> str:
    """20260910143000.export.CSV -> 20260910143000"""
    return file_path.split("/")[-1].split(".")[0]

"""Acceptance checks for the lakehouse over a date range.

Prints one PASS/FAIL line per check and exits 1 if anything fails, so the same script can
gate a backfill, a CI run or a monitoring job. Read-only.

Sections:
  COMPLETENESS    every day has the expected 15-minute slots; Bronze slots == landing files
  INTEGRITY       unique keys, no null keys/timestamps, malformed GKG rows within tolerance,
                  value ranges
  RECONCILIATION  Bronze -> Silver -> Gold tie out exactly
  QUALITY         GKG match rate above a threshold

Not included yet: comparing raw landing-file line counts with Bronze row counts (needs a
read of every CSV; will be an opt-in --deep check once the checks below are proven).
"""
import argparse
import sys
from datetime import datetime, timedelta

from pyspark.sql import SparkSession, functions as F

from bronze_ingest import MENTIONS_KEY
from utils import partner_country

LANDING = "s3a://gdelt-landing"

BRONZE = {
    "events": {"table": "local.bronze.events", "ts": "date_added_ts", "key": ["GLOBALEVENTID"]},
    "mentions": {"table": "local.bronze.mentions", "ts": "mention_ts", "key": MENTIONS_KEY},
    "gkg": {"table": "local.bronze.gkg", "ts": "gkg_ts", "key": ["GKGRECORDID"]},
}


class Report:
    def __init__(self):
        self.results = []

    def check(self, name: str, ok: bool, detail: str = ""):
        self.results.append((name, ok))
        print(f"{'PASS' if ok else 'FAIL'}  {name}" + (f"  -- {detail}" if detail else ""), flush=True)


def india_condition():
    return (
        (F.col("ActionGeo_CountryCode") == "IN")
        | (F.col("Actor1CountryCode") == "IND")
        | (F.col("Actor2CountryCode") == "IND")
    )


def bronze_slots_by_day(spark, table: str, ts_col: str, start_ts: str, end_ts: str) -> dict:
    rows = spark.sql(f"""
        SELECT date({ts_col}) AS day, COUNT(DISTINCT {ts_col}) AS slots
        FROM {table}
        WHERE {ts_col} >= TIMESTAMP '{start_ts}' AND {ts_col} < TIMESTAMP '{end_ts}'
        GROUP BY 1
    """).collect()
    return {r["day"].strftime("%Y%m%d"): r["slots"] for r in rows}


def landing_files_by_day(spark, stream: str) -> dict:
    path = spark._jvm.org.apache.hadoop.fs.Path(f"{LANDING}/{stream}/")
    fs = path.getFileSystem(spark._jsc.hadoopConfiguration())
    counts = {}
    for status in fs.listStatus(path):
        name = status.getPath().getName()          # e.g. 20260901000000.csv
        if len(name) >= 8 and name[:8].isdigit():
            counts[name[:8]] = counts.get(name[:8], 0) + 1
    return counts


def main():
    parser = argparse.ArgumentParser(description="Lakehouse acceptance checks.")
    parser.add_argument("--start-date", default="20260901", help="YYYYMMDD, inclusive")
    parser.add_argument("--end-date", default="20260930", help="YYYYMMDD, inclusive")
    parser.add_argument("--slots-per-day", type=int, default=96)
    parser.add_argument("--max-missing-slots", type=int, default=0,
                        help="Total missing slots tolerated per stream (GDELT-side gaps).")
    parser.add_argument("--max-malformed-gkg", type=int, default=10,
                        help="Tolerated GKG rows with a malformed record id / NULL timestamp.")
    parser.add_argument("--min-match-rate", type=float, default=0.95,
                        help="Minimum share of Silver mentions that found a GKG record.")
    args = parser.parse_args()

    start = datetime.strptime(args.start_date, "%Y%m%d")
    end = datetime.strptime(args.end_date, "%Y%m%d")
    if start > end:
        parser.error("--start-date cannot be after --end-date")
    days = []
    d = start
    while d <= end:
        days.append(d.strftime("%Y%m%d"))
        d += timedelta(days=1)
    start_ts = start.strftime("%Y-%m-%d %H:%M:%S")
    end_ts = (end + timedelta(days=1)).strftime("%Y-%m-%d %H:%M:%S")

    spark = (
        SparkSession.builder.appName("verify-all")
        .config("spark.executor.memory", "4g")
        .getOrCreate()
    )
    report = Report()

    print(f"=== COMPLETENESS ({args.start_date} -> {args.end_date}, {len(days)} days) ===")
    for stream, cfg in BRONZE.items():
        by_day = bronze_slots_by_day(spark, cfg["table"], cfg["ts"], start_ts, end_ts)
        short = {x: by_day.get(x, 0) for x in days if by_day.get(x, 0) < args.slots_per_day}
        extra = {x: by_day[x] for x in days if by_day.get(x, 0) > args.slots_per_day}
        missing = sum(args.slots_per_day - n for n in short.values())
        ok = missing <= args.max_missing_slots and not extra
        detail = f"{len(days) - len(short)}/{len(days)} full days, {missing} slots missing"
        if short:
            detail += f", short days (first 5): {list(short.items())[:5]}"
        if extra:
            detail += f", days with too many slots: {list(extra.items())[:5]}"
        report.check(f"{stream}: every day has {args.slots_per_day} slots", ok, detail)

        landing = landing_files_by_day(spark, stream)
        diff = {x: (landing.get(x, 0), by_day.get(x, 0)) for x in days
                if landing.get(x, 0) != by_day.get(x, 0)}
        detail = "all days match" if not diff else \
            f"{len(diff)} days differ (day: (landing, bronze)), first 5: {list(diff.items())[:5]}"
        report.check(f"{stream}: landing files == Bronze slots per day", not diff, detail)

    print("=== INTEGRITY ===")
    for stream, cfg in BRONZE.items():
        df = spark.table(cfg["table"])
        total = df.count()
        distinct = df.select(*cfg["key"]).distinct().count()
        report.check(f"{stream}: key is unique", total == distinct,
                     f"total={total} distinct={distinct}")
        nulls = df.filter(" OR ".join(f"{c} IS NULL" for c in cfg["key"])).count()
        report.check(f"{stream}: no NULL key columns", nulls == 0, f"null_key_rows={nulls}")

    for stream in ("events", "mentions"):
        cfg = BRONZE[stream]
        n = spark.table(cfg["table"]).filter(F.col(cfg["ts"]).isNull()).count()
        report.check(f"{stream}: no NULL {cfg['ts']}", n == 0, f"null_rows={n}")

    gkg = spark.table(BRONZE["gkg"]["table"])
    null_ts = gkg.filter(F.col("gkg_ts").isNull()).count()
    report.check("gkg: NULL gkg_ts within tolerance", null_ts <= args.max_malformed_gkg,
                 f"null_ts_rows={null_ts}, tolerance={args.max_malformed_gkg}")
    malformed = gkg.filter(~F.col("GKGRECORDID").rlike("^[0-9]{14}-")).count()
    report.check("gkg: malformed GKGRECORDID within tolerance", malformed <= args.max_malformed_gkg,
                 f"malformed_ids={malformed}, tolerance={args.max_malformed_gkg} "
                 f"(assumes ids look like 14 digits + '-')")

    out_of_range = spark.table(BRONZE["events"]["table"]).filter(
        (F.col("GoldsteinScale") < -10) | (F.col("GoldsteinScale") > 10)
        | (F.col("QuadClass").isNull() | ~F.col("QuadClass").isin(1, 2, 3, 4))
        | (F.abs(F.col("AvgTone")) > 100)
        | (F.col("ActionGeo_Lat") < -90) | (F.col("ActionGeo_Lat") > 90)
        | (F.col("ActionGeo_Long") < -180) | (F.col("ActionGeo_Long") > 180)
    ).count()
    report.check("events: GoldsteinScale/QuadClass/AvgTone/lat/long in range", out_of_range == 0,
                 f"out_of_range_rows={out_of_range}")

    print("=== RECONCILIATION ===")
    bronze_india_ids = (
        spark.table("local.bronze.events")
        .filter(india_condition())
        .filter(F.col("SOURCEURL").isNotNull() & F.col("GLOBALEVENTID").isNotNull())
        .select("GLOBALEVENTID").distinct()
    )
    silver_events = spark.table("local.silver.events_india")
    silver_mentions = spark.table("local.silver.mentions_enriched")

    expected = bronze_india_ids.count()
    actual = silver_events.count()
    report.check("Bronze -> Silver: events_india rows == India events in Bronze", expected == actual,
                 f"bronze_india_events={expected} silver={actual}")

    expected_pairs = (
        spark.table("local.bronze.mentions")
        .join(F.broadcast(bronze_india_ids), "GLOBALEVENTID", "inner")
        .select("GLOBALEVENTID", "MentionIdentifier").distinct().count()
    )
    actual = silver_mentions.count()
    report.check("Bronze -> Silver: mentions_enriched rows == distinct (event, article) pairs",
                 expected_pairs == actual, f"bronze_pairs={expected_pairs} silver={actual}")

    orphans = silver_mentions.join(silver_events.select("GLOBALEVENTID"), "GLOBALEVENTID", "left_anti").count()
    report.check("Silver: every mention belongs to a Silver event", orphans == 0,
                 f"orphan_mentions={orphans}")

    partner = partner_country(F.col("Actor1CountryCode"), F.col("Actor2CountryCode"))
    expected = silver_events.filter(partner.isNotNull() & (partner != "")).count()
    actual = spark.table("local.gold.india_bilateral_sentiment_daily") \
        .agg(F.sum("event_count")).first()[0] or 0
    report.check("Silver -> Gold: bilateral event_count sums to the Silver rows it covers",
                 expected == actual, f"silver_rows={expected} gold_sum={actual}")

    expected = silver_events.filter(
        (F.col("ActionGeo_CountryCode") == "IN") & F.col("ActionGeo_ADM1Code").isNotNull()
    ).count()
    actual = spark.table("local.gold.india_state_conflict_daily") \
        .agg(F.sum("event_count")).first()[0] or 0
    report.check("Silver -> Gold: state-conflict event_count sums to the Silver rows it covers",
                 expected == actual, f"silver_rows={expected} gold_sum={actual}")

    gold_mentions = spark.table("local.gold.india_event_mention_counts")
    expected = silver_mentions.count()
    actual = gold_mentions.agg(F.sum("total_mentions")).first()[0] or 0
    report.check("Silver -> Gold: total_mentions sums to Silver mention rows", expected == actual,
                 f"silver_rows={expected} gold_sum={actual}")

    expected = silver_mentions.select("GLOBALEVENTID").distinct().count()
    actual = gold_mentions.count()
    report.check("Silver -> Gold: one mention-count row per event", expected == actual,
                 f"silver_events_with_mentions={expected} gold_rows={actual}")

    print("=== QUALITY ===")
    total = silver_mentions.count()
    matched = silver_mentions.filter(F.col("V2Themes").isNotNull()).count()
    rate = matched / total if total else 0.0
    report.check("Silver: GKG match rate above threshold", rate >= args.min_match_rate,
                 f"match_rate={rate:.1%} threshold={args.min_match_rate:.0%} ({matched}/{total})")

    failed = [name for name, ok in report.results if not ok]
    print()
    print("=" * 60)
    print(f"SUMMARY: {len(report.results) - len(failed)} passed, {len(failed)} failed")
    for name in failed:
        print(f"  FAILED: {name}")
    print("=" * 60)

    spark.stop()
    if failed:
        sys.exit(1)


if __name__ == "__main__":
    main()

COPY spark/jobs ./jobs
COPY dagster/gdelt_pipeline ./gdelt_pipeline
COPY dagster/workspace.yaml ./workspace.yaml

# No ENTRYPOINT here on purpose.
#
# This used to be:
#   ENTRYPOINT ["dagster", "api", "grpc", "-h", "0.0.0.0", "-p", "4000", "-m", "gdelt_pipeline"]
#
# That's what broke `materialize all`. DockerRunLauncher creates ephemeral
# run containers straight from this image via the Docker Engine API,
# bypassing docker-compose entirely -- so your compose-level `entrypoint:`
# overrides never reach those containers. Docker always runs
# ENTRYPOINT + CMD together, and only CMD can be overridden without also
# explicitly overriding ENTRYPOINT. So every run container was actually
# executing:
#   dagster api grpc -h 0.0.0.0 -p 4000 -m gdelt_pipeline  dagster api execute_run {...}
# as one command, and Click rejected the second half as unexpected extra
# arguments -- the run container crashed on start, every time, which is
# why materialize all just sat there stuck.
#
# The grpc command now lives ONLY in docker-compose.yml's `entrypoint:`
# override for the dagster-user-code service. Run containers spawned by
# DockerRunLauncher now get no fixed entrypoint to collide with, so
# "dagster api execute_run ..." runs cleanly on its own.

      # which is why the run looked "stuck" in Dagit instead of failing.
      - DAGSTER_POSTGRES_USER
      - DAGSTER_POSTGRES_PASSWORD
      - DAGSTER_POSTGRES_DB
      - AWS_ACCESS_KEY_ID
      - AWS_SECRET_ACCESS_KEY
      - CATALOG_URI
      - WAREHOUSE_BUCKET
      - AWS_REGION
      - SPARK_MASTER_URL
      - SLACK_WEBHOOK_URL
      - PYTHONPATH
    container_kwargs:
      # lakehouse-net has no IPv6 route -- data.gdeltproject.org (and
      # possibly other external hosts) publish AAAA records, so without
      # this, every outbound request wastes a connection attempt on an
      # unreachable IPv6 address before falling back to IPv4. Disabling
      # IPv6 here makes that fallback unnecessary instead of relying on
      # it working correctly every time.
      sysctls:
        net.ipv6.conf.all.disable_ipv6: "1"
      volumes:
        # named volume -- fine for ephemeral containers too
        - "spark_jars:/opt/iceberg-jars:ro"

# Events/Mentions: 99.9% arrive within 8 min of slot per the arrival analysis -- a short,
# fast retry is correct here. No reason to wait on GKG's much longer tail.
FAST_RETRY = RetryPolicy(max_retries=3, delay=20, backoff=Backoff.EXPONENTIAL, jitter=Jitter.PLUS_MINUS)
RETRY = RetryPolicy(max_retries=3, delay=30, backoff=Backoff.EXPONENTIAL, jitter=Jitter.PLUS_MINUS)


@asset(retry_policy=FAST_RETRY)
def bronze_events(context: AssetExecutionContext, spark: IcebergSparkResource):
    ts = get_latest_timestamps()["events"]
    ok = fetch_and_upload(ts, "events", s3_client(), max_retries=3, backoff_base=20)
    if not ok:
        raise Exception(f"events file missing for {ts}")
    session = spark.get_spark()
    new_rows, total_rows = ingest_events_batch(session, f"s3a://gdelt-landing/events/{ts}.csv")
    context.add_output_metadata({"timestamp": ts, "new_rows": new_rows, "batch_rows": total_rows})
    session.stop()

"""Ported from scripts/download_gdelt.py, adapted for in-container use
(minio:9000 instead of localhost:9000) plus the per-stream retry we
added after discovering GKG's slower-processing 404 behavior."""

import io
import os
import time
import zipfile
import requests
import boto3

MANIFEST_URL = "http://data.gdeltproject.org/gdeltv2/lastupdate.txt"

BASE_URL = "http://data.gdeltproject.org/gdeltv2"

STREAM_SUFFIX = {
    "events": "export.CSV.zip",
    "mentions": "mentions.CSV.zip",
    "gkg": "gkg.csv.zip",
}


def get_latest_timestamps() -> dict:
    resp = requests.get(
        MANIFEST_URL,
        timeout=30,
    )
    resp.raise_for_status()
    latest = {}
    for line in resp.text.strip().split("\n"):
        _, _, url = line.split()
        for stream, suffix in STREAM_SUFFIX.items():
            if url.endswith(suffix):
                latest[stream] = (
                    url.rsplit("/", 1)[-1]
                    .split(".")[0]
                )
    return latest


def s3_client():
    return boto3.client(
        "s3",
        endpoint_url="http://minio:9000",
        aws_access_key_id=os.environ["AWS_ACCESS_KEY_ID"],
        aws_secret_access_key=os.environ["AWS_SECRET_ACCESS_KEY"],
    )


def fetch_and_upload(
    timestamp: str, stream: str, client, max_retries=3, backoff_base=30
) -> bool:
    url = f"{BASE_URL}/{timestamp}.{STREAM_SUFFIX[stream]}"
    for attempt in range(1, max_retries + 1):
        resp = requests.get(url, timeout=60)
        if resp.status_code == 404:
            if attempt < max_retries:
                time.sleep(backoff_base * attempt)
                continue
            return False
        resp.raise_for_status()
        with zipfile.ZipFile(io.BytesIO(resp.content)) as z:
            data = z.read(z.namelist()[0])
        client.put_object(
            Bucket="gdelt-landing", Key=f"{stream}/{timestamp}.csv", Body=data
        )
        return True
    return False

import json

import requests

from dagster import (
    sensor,
    RunRequest,
    SensorEvaluationContext,
    DefaultSensorStatus,
    SkipReason,
)

from .gdelt_fetch import BASE_URL, get_latest_timestamps


@sensor(
    job_name="gdelt_gkg_ingest",
    minimum_interval_seconds=60,
    default_status=DefaultSensorStatus.RUNNING,
)
def gkg_availability_sensor(context: SensorEvaluationContext):
    """
    GKG availability sensor.

    Behavior:

    1. Discover the latest GDELT GKG timestamp.
    2. Once a timestamp becomes the pending timestamp, keep retrying
       that timestamp until the file is available.
    3. Do NOT move to a newer timestamp while the pending timestamp
       is still unavailable.
    4. Once the pending file is available, request the ingestion run.
    5. Only after the run request is created do we clear the pending
       timestamp.
    """

    # ---------------------------------------------------------
    # 1. Read sensor state
    # ---------------------------------------------------------

    raw_cursor = context.cursor

    processed_ts = None
    pending_ts = None

    if raw_cursor:
        try:
            state = json.loads(raw_cursor)
            if not isinstance(state, dict):
                raise ValueError("legacy cursor")

            processed_ts = state.get("processed")
            pending_ts = state.get("pending")

        except (json.JSONDecodeError, ValueError):
            processed_ts = raw_cursor.removesuffix(":alerted")
            pending_ts = None

    context.log.info(
        f"GKG sensor state: "
        f"processed={processed_ts}, "
        f"pending={pending_ts}"
    )

    # ---------------------------------------------------------
    # 2. Get latest timestamp from GDELT
    # ---------------------------------------------------------

    try:
        latest = get_latest_timestamps()
        latest_gkg_ts = latest["gkg"]

    except requests.RequestException as exc:
        context.log.warning(
            f"Could not reach GDELT manifest: {exc}"
        )

        return SkipReason(
            "GDELT manifest temporarily unavailable; "
            "will retry on the next sensor tick."
        )

    except KeyError:
        context.log.error(
            f"GDELT manifest did not contain a GKG timestamp: {latest}"
        )

        return SkipReason(
            "GDELT manifest did not contain a GKG timestamp; "
            "will retry on the next sensor tick."
        )

    context.log.info(
        f"GDELT latest GKG timestamp: {latest_gkg_ts}"
    )

    # ---------------------------------------------------------
    # 3. Determine which timestamp we should check
    # ---------------------------------------------------------
    #
    # IMPORTANT:
    #
    # If we already have a pending timestamp, KEEP checking it.
    #
    # Example:
    #
    # pending = 17:45
    # latest  = 18:00
    #
    # We still check 17:45.
    #

    if pending_ts:
        target_ts = pending_ts

        context.log.info(
            f"Retrying pending GKG timestamp: {target_ts}"
        )

    else:

        # Nothing pending.
        #
        # If latest == processed, we've already handled it.

        if processed_ts == latest_gkg_ts:
            return SkipReason(
                f"GKG {latest_gkg_ts} already processed/requested; "
                "waiting for next GDELT update."
            )

        # Start tracking the latest timestamp.
        target_ts = latest_gkg_ts

        pending_ts = target_ts

        context.log.info(
            f"New pending GKG timestamp: {target_ts}"
        )

    # ---------------------------------------------------------
    # 4. Check the target GKG file
    # ---------------------------------------------------------

    url = f"{BASE_URL}/{target_ts}.gkg.csv.zip"

    context.log.info(
        f"Checking GKG file: {url}"
    )

    try:
        response = requests.get(
            url,
            stream=True,
            allow_redirects=True,
            timeout=15,
        )

        status_code = response.status_code
        response.close()

    except requests.RequestException as exc:

        context.log.warning(
            f"Could not reach GDELT GKG file: {exc}"
        )

        # IMPORTANT:
        #
        # Keep pending_ts in the cursor so we retry the SAME
        # timestamp next time.

        context.update_cursor(
            json.dumps(
                {
                    "processed": processed_ts,
                    "pending": target_ts,
                }
            )
        )

        return SkipReason(
            f"GDELT GKG file check failed temporarily for "
            f"{target_ts}; will retry."
        )

    # ---------------------------------------------------------
    # 5. File is available
    # ---------------------------------------------------------

    if status_code == 200:

        context.log.info(
            f"GKG file is available: {target_ts}"
        )

        # Store this timestamp as processed.
        #
        # Clear pending because we are now creating the run.

        context.update_cursor(
            json.dumps(
                {
                    "processed": target_ts,
                    "pending": None,
                }
            )
        )

        return RunRequest(
            run_key=target_ts,
            run_config={
                "ops": {
                    "bronze_gkg": {
                        "config": {
                            "timestamp": target_ts,
                        }
                    }
                }
            },
        )

    # ---------------------------------------------------------
    # 6. File is not published yet
    # ---------------------------------------------------------

    if status_code == 404:

        context.log.info(
            f"GKG {target_ts} is not published yet. "
            f"Keeping it as pending."
        )

        # VERY IMPORTANT:
        #
        # Do NOT replace target_ts with the newer GDELT timestamp.
        #
        # Keep the current timestamp as pending.

        context.update_cursor(
            json.dumps(
                {
                    "processed": processed_ts,
                    "pending": target_ts,
                }
            )
        )

        return SkipReason(
            f"GKG {target_ts} is not published yet; "
            f"will keep retrying this timestamp."
        )

    # ---------------------------------------------------------
    # 7. Other HTTP status
    # ---------------------------------------------------------

    context.log.warning(
        f"GKG {target_ts} returned HTTP {status_code}; "
        f"keeping it as pending."
    )

    context.update_cursor(
        json.dumps(
            {
                "processed": processed_ts,
                "pending": target_ts,
            }
        )
    )

    return SkipReason(
        f"GKG {target_ts} returned HTTP {status_code}; "
        f"will retry."
    )

"""Small helper for building DataFrames in tests without spelling out every column."""


def make_df(spark, schema, defaults, rows):
    """Build a DataFrame from a DDL-style schema string. Each row is a dict that only needs
    the columns it cares about; the rest come from `defaults`."""
    cols = [part.split()[0] for part in schema.split(",")]
    data = [tuple({**defaults, **row}[c] for c in cols) for row in rows]
    return spark.createDataFrame(data, schema)

    assert len(chunks) == 5
    assert chunks[0] == (date(2026, 9, 1), date(2026, 9, 7))
    assert chunks[-1] == (date(2026, 9, 29), date(2026, 9, 30))


@pytest.mark.parametrize("chunk_days", [1, 3, 7, 10, 45])
def test_chunks_cover_range_exactly_once(chunk_days):
    start, end = date(2026, 9, 1), date(2026, 10, 17)
    chunks = list(backfill.split_into_chunks(start, end, chunk_days))
    assert chunks[0][0] == start and chunks[-1][1] == end
    for (_, prev_end), (next_start, _) in zip(chunks, chunks[1:]):
        assert (next_start - prev_end).days == 1          # no gap, no overlap
    assert all((b - a).days + 1 <= chunk_days for a, b in chunks)
    assert sum((b - a).days + 1 for a, b in chunks) == (end - start).days + 1

def make_args(**overrides):
    base = dict(
        start_date=date(2026, 9, 1),
        end_date=date(2026, 9, 30),
        workers=6,
        retries=4,
        chunk_days=7,
        streams=["events", "mentions", "gkg"],
        download_only=False,
        ingest_only=False,
    )
    base.update(overrides)
    return SimpleNamespace(**base)


def test_validate_arguments_accepts_good_args():
    backfill.validate_arguments(make_args())


@pytest.mark.parametrize(
    "overrides, message",
    [
        (dict(start_date=date(2026, 10, 1), end_date=date(2026, 9, 1)), "cannot be after"),
        (dict(workers=0), "--workers"),
        (dict(retries=0), "--retries"),
        (dict(chunk_days=0), "--chunk-days"),
        (dict(streams=["events", "bogus"]), "Invalid streams"),
        (dict(download_only=True, ingest_only=True), "cannot be used together"),
    ],
)
def test_validate_arguments_rejects_bad_args(overrides, message):
    with pytest.raises(SystemExit) as exc:
        backfill.validate_arguments(make_args(**overrides))
    assert message in str(exc.value)


# --------------------------------------------------------------------------- acquire_lock
def test_second_backfill_cannot_take_the_lock(tmp_path, monkeypatch):
    monkeypatch.setattr(backfill, "LOCK_PATH", str(tmp_path / ".backfill.lock"))
    first = backfill.acquire_lock()
    try:
        with pytest.raises(SystemExit) as exc:
            backfill.acquire_lock()
        assert "already running" in str(exc.value)
    finally:
        first.close()


def test_lock_is_released_when_the_holder_closes(tmp_path, monkeypatch):
    """The lock is an OS flock, so a leftover .backfill.lock FILE does not block anything."""
    monkeypatch.setattr(backfill, "LOCK_PATH", str(tmp_path / ".backfill.lock"))
    backfill.acquire_lock().close()
    again = backfill.acquire_lock()      # must not raise
    again.close()

"""Tests for spark/jobs/utils.py -- the logic shared by Silver, Gold and verify_all."""
import pytest

from utils import conflict_severity, extract_batch_id, normalize_url, partner_country


# --------------------------------------------------------------------------- normalize_url
@pytest.mark.parametrize(
    "raw, expected",
    [
        # case, whitespace, fragment
        ("  https://Site.com/Story  ", "https://site.com/story"),
        ("https://site.com/a#comments", "https://site.com/a"),
        # tracking parameters are dropped ...
        ("https://site.com/a?utm_source=x", "https://site.com/a"),
        ("https://site.com/a?utm_medium=1&utm_campaign=2", "https://site.com/a"),
        ("https://site.com/a?fbclid=abc", "https://site.com/a"),
        # ... in any position, without breaking the remaining query string
        ("https://site.com/a?utm_source=x&id=5", "https://site.com/a?id=5"),
        ("https://site.com/a?id=5&utm_source=x", "https://site.com/a?id=5"),
        ("https://site.com/a?id=5&fbclid=z&p=2", "https://site.com/a?id=5&p=2"),
        # parameters that identify the article are kept
        ("https://site.com/story.php?id=5", "https://site.com/story.php?id=5"),
        ("https://site.com/?p=123", "https://site.com/?p=123"),
    ],
)
def test_normalize_url(spark, raw, expected):
    df = spark.createDataFrame([(raw,)], ["url"])
    assert df.select(normalize_url(F.col("url"))).first()[0] == expected


def test_normalize_url_keeps_different_articles_apart(spark):
    """The bug this guards against: stripping the whole query string used to merge
    story.php?id=1 and story.php?id=2, so one GKG row got attached to both."""
    df = spark.createDataFrame(
        [("https://site.com/story.php?id=1",), ("https://site.com/story.php?id=2",)], ["url"]
    )
    distinct = df.select(normalize_url(F.col("url")).alias("n")).distinct().count()
    assert distinct == 2


def test_normalize_url_merges_same_article_with_tracking_noise(spark):
    df = spark.createDataFrame(
        [("https://site.com/a?id=1",), ("https://site.com/a?id=1&utm_source=tw",)], ["url"]
    )
    assert df.select(normalize_url(F.col("url")).alias("n")).distinct().count() == 1


def test_normalize_url_null_stays_null(spark):
    df = spark.createDataFrame([(None,)], "url string")
    assert df.select(normalize_url(F.col("url"))).first()[0] is None
